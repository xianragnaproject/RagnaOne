# eventMacros.txt — Job Master + Knight leveling route
# Novice → Swordman → Knight  |  same map progression as Assassin pack

# Job Master @ Prontera 150,180 (live Job Master#ep2)
# First jobs: Swordman, Magician, Archer, Acolyte, Merchant, Thief
#   → Swordman = r0
# Swordman 2nd: Knight, Crusader
#   → Knight = r0

automacro Ragna_Job_ToSwordman {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Prontera Job Master → Swordman
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c c c r0 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Swordman requested
	}
}

# Equip any available weapon/armor from inventory if slots empty (shops missing on this host)
automacro Ragna_EquipIfEmpty {
	BaseLevel >= 1
	timeout 45
	exclusive 1
	Eval (!$char->{equipment}{rightHand} || !$char->{equipment}{armor})
	call {
		[
			log [Ragna] Empty gear slot - equipping best from inventory
			do eq
			pause 0.5
		]
	}
}

	}
}

# Yank bots that wandered onto wrong fields back to lockMap
automacro Ragna_TownSell {
	ConfigKeyNot lockMap none
	IsInMapAndCoordinate prontera,morocc,geffen,payon,alberta,prt_in,payon_in01
	timeout 20
	exclusive 1
	Eval (Misc::percent_weight($char) >= 10) && !(AI::inQueue("sellAuto") || AI::inQueue("storageAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","storageAuto","buyAuto"))
	call {
		[
			log [Ragna] Town weight >= 10% - selling junk
			do conf attackAuto 2
			do conf route_randomWalk 1
			do eval AI::queue("sellAuto");
			pause 1
			do eq
		]
	}
}

# Set Prontera save once so future deaths skip Izlude spawn

	IsInMapAndCoordinate prontera

			do conf saveMap_y 187
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
		]
	}
}

# Apply Payon NPC coords to every buyAuto_* entry (called by Ragna_ShopsPayon)
sub Ragna_ApplyPayonShops {
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		my ($npc, $stand);
		if ($item =~ /Potion|Magnifier|Wing|^Arrow$/i) {
			$npc = "payon_in01 5 49";
			$stand = "payon_in01 7 49";
		} elsif ($item =~ /Cotton Shirt|Jacket/) {
			$npc = "payon_in01 7 119";
			$stand = "payon_in01 10 119";
		} else {
			$npc = "payon_in01 15 119";
			$stand = "payon_in01 15 117";
		}
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand");
	}
	return 1;
}

# At Base 12+ switch sell/buy NPCs from Prontera to Payon
automacro Ragna_ShopsPayon {
	BaseLevel >= 12
	ConfigKeyNot ragnaShopsPayon 1
	timeout 60
	exclusive 1
	call {
		[
			log [Ragna] Lv12+ — switching sell/buy NPCs to Payon
			do conf sellAuto_npc payon_in01 5 49
			do conf sellAuto_standpoint payon_in01 7 49
			do eval Ragna_ApplyPayonShops()
			do conf ragnaShopsPayon 1
			log [Ragna] Sell/buy now use Payon shops - tool armor weapon
		]
	}
}

	IsInMapAndCoordinate izlude
	timeout 35
	exclusive 1

}

macro EscapeIzlude {
	log [Ragna] Escaping Izlude via west portal to prt_fild08
	do conf attackAuto 0
	do conf route_randomWalk 0
	do conf lockMap none
	pause 1
	do move 30 78
	pause 14
	do conf attackAuto 2
	do conf route_randomWalk 1
	do conf lockMap prt_fild08
}
