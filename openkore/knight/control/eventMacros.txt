# eventMacros.txt — Job Master + Knight leveling route
# load eventMacro

# Job Master @ Prontera 153,193 (Euphy)
# First jobs: Swordman=r0, Magician, Archer, Acolyte, Merchant, Thief
# Swordman 2nd: Knight=r0, Crusader=r1

automacro Ragna_Job_ToSwordman {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Job Master → Swordman
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 153 193
		pause 2
		do talknpc 153 193 c r0 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Swordman requested
	}
}

}

	timeout 420
	call {
		log [RagnaOne] Knight short idle pause...

}

	ConfigKeyNotIn isDead 0
	delay 3
	run-once 0
	call {
		log [RagnaOne] Died — recovering in town
		pause 5..12
	}
}
