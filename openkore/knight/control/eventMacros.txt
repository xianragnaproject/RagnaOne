# eventMacros.txt - Job Master + Knight leveling route
# Novice → Swordman → Knight  |  same map progression as Assassin pack

# Job Master @ Prontera 150,180 (live Job Master#ep2)
# First jobs: Swordman, Magician, Archer, Acolyte, Merchant, Thief
#   → Swordman = r0
# Swordman 2nd: Knight, Crusader
#   → Knight = r0

automacro Ragna_Job_ToSwordman {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Prontera Job Master → Swordman
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c c c r0 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Swordman requested
	}
}

# Unequipped: force sell/buy restock; pause hunt return until gear is on
automacro Ragna_UnequippedRestock {
	BaseLevel >= 1
	ConfigKeyNot ragnaGearRestock 1
	timeout 75
	exclusive 1
	Eval ((!$char->{equipment}{rightHand}) || (!$char->{equipment}{armor})) && (Misc::percent_weight($char) >= 8) && !(AI::inQueue("sellAuto") || AI::inQueue("storageAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","storageAuto","buyAuto"))
	call {
		[
			log Ragna Empty gear + junk - Prontera sell/buy restock
			do conf ragnaGearRestock 1
			do conf ragnaLockMapBackup @config(lockMap)
			do conf ragnaShopsPayon 0
			do conf lockMap none
			do conf attackAuto 0
			do conf route_randomWalk 0
			do eval Ragna_ApplyPronteraShops()
			do eval AI::queue("sellAuto");
			pause 2
		]
	}
}

automacro Ragna_GearRestockDone {
	ConfigKey ragnaGearRestock 1
	timeout 30
	exclusive 1
	Eval ($char->{equipment}{rightHand} && $char->{equipment}{armor})
	call {
		[
			log Ragna Gear restocked - resume hunt
			do conf ragnaGearRestock 0
			do conf lockMap @config(ragnaLockMapBackup)
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf ragnaShopsPayon 0
			do eq
		]
	}
}

# If restock flag set but sell/buy not running and still naked, re-queue sell
automacro Ragna_GearRestockRetry {
	ConfigKey ragnaGearRestock 1
	timeout 90
	exclusive 1
	Eval ((!$char->{equipment}{rightHand}) || (!$char->{equipment}{armor})) && !(AI::inQueue("sellAuto") || AI::inQueue("storageAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","storageAuto","buyAuto"))
	call {
		[
			log Ragna Gear restock retry - queue sellAuto again
			do conf lockMap none
			do conf attackAuto 0
			do conf route_randomWalk 0
			do eval Ragna_ApplyPronteraShops()
			do eval AI::queue("sellAuto");
		]
	}
}

# Apply Payon NPC coords to every buyAuto_* entry (called by Ragna_ShopsPayon)

sub Ragna_ApplyPronteraShops {
	Commands::run("conf sellAuto_npc prt_in 126 76");
	Commands::run("conf sellAuto_standpoint prt_in 126 74");
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		my ($npc, $stand);
		if ($item =~ /Potion|Magnifier|Wing|^Arrow$/i) {
			$npc = "prt_in 126 76";
			$stand = "prt_in 126 74";
		} elsif ($item =~ /Cotton Shirt|Jacket/) {
			$npc = "prt_in 172 132";
			$stand = "prt_in 172 134";
		} else {
			$npc = "prt_in 172 130";
			$stand = "prt_in 172 128";
		}
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand");
	}
	return 1;
}

sub Ragna_ApplyPayonShops {
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		my ($npc, $stand);
		if ($item =~ /Potion|Magnifier|Wing|^Arrow$/i) {
			$npc = "payon_in01 5 49";
			$stand = "payon_in01 7 49";
		} elsif ($item =~ /Cotton Shirt|Jacket/) {
			$npc = "payon_in01 7 119";
			$stand = "payon_in01 10 119";
		} else {
			$npc = "payon_in01 15 119";
			$stand = "payon_in01 15 117";
		}
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand");
	}
	return 1;
}

# At Base 12+ switch sell/buy NPCs from Prontera to Payon
automacro Ragna_ShopsPayon {
	BaseLevel >= 12
	ConfigKeyNot ragnaShopsPayon 1
	timeout 60
	exclusive 1
	call {
		[
			log Ragna Lv12+ - switching sell/buy NPCs to Payon
			do conf sellAuto_npc payon_in01 5 49
			do conf sellAuto_standpoint payon_in01 7 49
			do eval Ragna_ApplyPayonShops()
			do conf ragnaShopsPayon 1
			log Ragna Sell/buy now use Payon shops - tool armor weapon
		]
	}
}
