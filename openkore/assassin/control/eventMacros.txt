# eventMacros.txt — Job Master + Assassin leveling route
# Requires OpenKore eventMacro plugin (bundled in modern OpenKore).
# Enable with: load eventMacro

# ============================================================
# Job Master @ Prontera 150,180 (stock Hercules Euphy Job Master)
# Menu order may differ on custom scripts — tune r# if needed.
# First jobs often: Swordman, Magician, Archer, Acolyte, Merchant, Thief
#   → Thief = r5
# Thief 2nd jobs often: Assassin, Rogue
#   → Assassin = r0
# ============================================================

automacro Ragna_Job_ToThief {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Prontera Job Master → Thief
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 153 193
		pause 2
		do talknpc 150 180 c r5 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Thief requested
	}
}

# Identify unidentified loot with Magnifier, then re-check equipAuto
automacro Ragna_IdentifyLoot {
	BaseLevel >= 1
	InInventory "Magnifier" > 0
	timeout 6
	exclusive 1
	call {
		$n = @eval(my $c=0; foreach my $it (@{$char->inventory->getItems()}) { $c++ if ($it && !$it->{identified}); } $c)
		if ($n < 1) {
			stop
		}
		[
			log [Ragna] Magnifier: identify loot then equip check
			do is Magnifier
			pause 1
			do identify 0
			pause 0.8
			do eq
			pause 0.5
			log [Ragna] Identify/equip pass done
		]
	}
}

		]
	}
}

automacro Ragna_AfterDeath {
	isAlive 0
	timeout 30
	exclusive 1
	call {
		[
			log [Ragna] Dead - no pause to avoid orphan macros
		]
	}
}

# ---- Town / Izlude / off-lockMap recovery (RagnaOne) ----
automacro Ragna_TownAfterDeath {
	ConfigKeyNot lockMap none
	IsInMapAndCoordinate prontera,morocc,geffen,payon,alberta
	timeout 25
	exclusive 1
	Eval !(AI::inQueue("sellAuto") || AI::inQueue("storageAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","storageAuto","buyAuto"))
	call {
		[
			log [Ragna] In town - resume lockMap
			do conf attackAuto 2
			do conf route_randomWalk 1
			pause 1
		]
	}
}

# Yank bots that wandered onto wrong fields back to lockMap
automacro Ragna_ReturnToLock {
	ConfigKeyNot lockMap none
	InLockMap 0
	timeout 40
	exclusive 1
	Eval !(AI::inQueue("sellAuto") || AI::inQueue("storageAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","storageAuto","buyAuto"))
	call {
		[
			log [Ragna] Not on lockMap - resume hunt route
			do conf attackAuto 2
			do conf route_randomWalk 1
			pause 2
		]
	}
}

# Set Prontera save once so future deaths skip Izlude spawn
automacro Ragna_SetSaveProntera {
	IsInMapAndCoordinate prontera
	ConfigKeyNot ragnaSaveProntera 1
	timeout 90
	exclusive 1
	call {
		[
			log [Ragna] Setting save point at Prontera Kafra
			do conf attackAuto 0
			do conf route_randomWalk 0
			do move 151 29
			pause 8
			do talknpc 151 29 c r0
			pause 4
			do conf ragnaSaveProntera 1
			do conf saveMap prontera
			do conf saveMap_x 155
			do conf saveMap_y 187
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
		]
	}
}

automacro Ragna_EscapeIzlude {
	IsInMapAndCoordinate izlude
	timeout 35
	exclusive 1
	call EscapeIzlude
}

macro EscapeIzlude {
	log [Ragna] Escaping Izlude via west portal to prt_fild08
	do conf attackAuto 0
	do conf route_randomWalk 0
	do conf lockMap none
	pause 1
	do move 30 78
	pause 14
	do conf attackAuto 2
	do conf route_randomWalk 1
	do conf lockMap prt_fild08
}
