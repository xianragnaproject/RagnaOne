# Hunter path: Novice → Archer → Hunter
# Job Master @ prontera 150,180

automacro Ragna_Job_ToArcher {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Job Master → Archer
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c c c r2 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Archer requested
	}
}
