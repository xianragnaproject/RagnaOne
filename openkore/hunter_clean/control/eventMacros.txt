# AssassinClean - Lv1-13: hunt / buy pots / sell loot
# No equipment shopping while Novice

automacro Ragna_SetSavePronteraSouth {
	IsInMapAndCoordinate prontera
	ConfigKeyNot ragnaSaveProntera 1
	timeout 90
	exclusive 1
	call {
		[
			log Ragna Setting save at Prontera south Kafra
			do conf attackAuto 0
			do conf route_randomWalk 0
			do move 151 29
			pause 8
			do talknpc 151 29 c r0
			pause 4
			do conf ragnaSaveProntera 1
			do conf saveMap prontera
			do conf saveMap_x 155
			do conf saveMap_y 50
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
			log Ragna Save set Prontera south - hunting prt_fild08
		]
	}
}

# Sell loot in town when overweight (sellAuto also triggers via itemsMaxWeight_sellOrStore)
automacro Ragna_TownSell {
	ConfigKey ragnaShop 1
	ConfigKeyNot lockMap none
	IsInMapAndCoordinate prontera,prt_in,payon,payon_in01
	timeout 25
	exclusive 1
	Eval (Misc::percent_weight($char) >= 18) && !(AI::inQueue("sellAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","buyAuto"))
	call {
		[
			log Ragna In town with weight - selling loot
			do conf attackAuto 2
			do conf route_randomWalk 1
			do eval AI::queue("sellAuto");
			pause 1
		]
	}
}

# Resume lockMap: Payon field after save, else Prontera field (novice era)
sub Ragna_ResumeHuntLock {
	my $map = ($config{ragnaSavePayon}) ? "pay_fild08" : "prt_fild08";
	Commands::run("conf lockMap $map");
	Commands::run("conf attackAuto 2");
	Commands::run("conf route_randomWalk 1");
	return 1;
}


# Enable gear buyAuto only after first job (weapons, armor, arrows).
# Novices must NOT buy equipment — pots/wings only.
sub Ragna_EnableGearBuy {
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		next if $item =~ /Potion|Wing|Magnifier/i;
		Commands::run("conf buyAuto_${i}_disabled 0");
	}
	return 1;
}

sub Ragna_ApplyPayonShops {
	Commands::run("conf sellAuto_npc payon_in01 5 49");
	Commands::run("conf sellAuto_standpoint payon_in01 7 49");
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		my ($npc, $stand);
		if ($item =~ /Potion|Magnifier|Wing|^Arrow$/i) {
			$npc = "payon_in01 5 49";
			$stand = "payon_in01 7 49";
		} elsif ($item =~ /Cotton Shirt|Jacket/) {
			$npc = "payon_in01 7 119";
			$stand = "payon_in01 10 119";
		} else {
			$npc = "payon_in01 15 119";
			$stand = "payon_in01 15 117";
		}
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand");
	}
	return 1;
}
