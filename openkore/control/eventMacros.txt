# eventMacros.txt — change hunting grounds by base level (normal player route)
# Requires OpenKore eventMacro plugin (bundled in modern OpenKore).
# Enable with: load eventMacro
# Or ensure automacro/eventMacro is active in your OpenKore build.

automacro Ragna_Lv1_11 {
	BaseLevel < 12
	run-once 1
	call {
		do conf lockMap prt_fild08
		do conf lockMap_randX 14
		do conf lockMap_randY 14
		log [RagnaOne] Lv1-11: Prontera Field 08 (porings/drops)
	}
}

# Occasional sit pause to look less robotic (idle heal already handles most)
automacro Ragna_RandomIdle {
	timeout 420
	call {
		log [RagnaOne] Taking a short idle pause...
		do sit
		pause 8..18
		do stand
	}
}

# After death, restock feeling
automacro Ragna_AfterDeath {
	ConfigKeyNotIn isDead 0
	delay 3
	run-once 0
	call {
		log [RagnaOne] Died — recovering in town before returning
		pause 5..12
	}
}
