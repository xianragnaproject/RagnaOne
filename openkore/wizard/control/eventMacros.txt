# Wizard path: Novice → Magician → Wizard
# Job Master @ prontera 150,180

automacro Ragna_Job_ToMagician {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Job Master → Magician
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c r1 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Magician requested
	}
}
