# AssassinClean - Lv1-13: hunt / buy pots / sell loot
# No equipment shopping while Novice

automacro Ragna_SetSavePronteraSouth {
	IsInMapAndCoordinate prontera
	ConfigKeyNot ragnaSaveProntera 1
	timeout 90
	exclusive 1
	call {
		[
			log Ragna Setting save at Prontera south Kafra
			do conf attackAuto 0
			do conf route_randomWalk 0
			do move 151 29
			pause 8
			do talknpc 151 29 c r0
			pause 4
			do conf ragnaSaveProntera 1
			do conf saveMap prontera
			do conf saveMap_x 155
			do conf saveMap_y 50
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
			log Ragna Save set Prontera south - hunting prt_fild08
		]
	}
}

# Sell loot in town when overweight (sellAuto also triggers via itemsMaxWeight_sellOrStore)
automacro Ragna_TownSell {
	ConfigKey ragnaShop 1
	ConfigKeyNot lockMap none
	IsInMapAndCoordinate prontera,prt_in
	timeout 25
	exclusive 1
	Eval (Misc::percent_weight($char) >= 18) && !(AI::inQueue("sellAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","buyAuto"))
	call {
		[
			log Ragna In town with weight - selling loot
			do conf attackAuto 2
			do conf route_randomWalk 1
			do eval AI::queue("sellAuto");
			pause 1
		]
	}
}

# After buy/sell finishes, leave prt_in before portal loop traps the bot
automacro Ragna_EscapePrtShop {
	ConfigKey ragnaShop 1
	IsInMapAndCoordinate prt_in
	InInventoryID 501 >= 3
	timeout 50
	exclusive 1
	Eval !(AI::inQueue("sellAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","buyAuto","NPC"))
	call {
		[
			log Ragna Leaving Prontera tool shop avoid portal loop
			do conf attackAuto 0
			do conf route_randomWalk 0
			pause 1
			do move 126 70
			pause 3
			do move prontera 160 70
			pause 14
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
		]
	}
}

# If still stuck in prt_in after escape attempt, Butterfly Wing out
automacro Ragna_WingOutPrtIn {
	ConfigKey ragnaShop 1
	IsInMapAndCoordinate prt_in
	InInventoryID 602 > 0
	timeout 90
	exclusive 1
	Eval !(AI::inQueue("sellAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","buyAuto","NPC"))
	call {
		[
			log Ragna Stuck in prt_in Butterfly Wing escape
			do conf attackAuto 0
			do conf route_randomWalk 0
			pause 1
			do is Butterfly Wing
			pause 4
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
		]
	}
}

# Job change Novice -> Thief at Job Lv 10
automacro Ragna_Job_ToThief {
	JobID 0
	JobLevel >= 10
	BaseLevel >= 10
	ConfigKeyNot ragnaJobThief 1
	timeout 120
	exclusive 1
	call {
		[
			log Ragna Job change to Thief at Prontera Job Master
			do conf attackAuto 0
			do conf route_randomWalk 0
			do conf lockMap none
			do move prontera 150 180
			pause 14
			do talknpc 150 180 c c c r1 c r0
			pause 5
			do conf ragnaJobThief 1
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf lockMap prt_fild08
			log Ragna Now Thief - back to prt_fild08
		]
	}
}

# ============================================================
# Field Scout#prt08a @ prt_fild08 160,370  (also prt08b at 180,370)
# Daily hunt quest (50x/day): kill 20 on prt_fild08 - Novice only
# Reward: 500 Base / 500 Job / 1000 Zeny / 20 Red Potions
# Quest ID 60202 | mission mob_id 10363 (map hunt counter)
# Accept:  talknpc 160 370 c c r0
# Turn-in: talknpc 160 370 c c
# Note: main label Field Scout#prt08 at 170,360 is often missing;
#       use prt08a. Must walk in range first or TalkNPC cannot see NPC.
# Disable with: conf ragnaFieldScoutOff 1
# ============================================================

automacro Ragna_FieldScout_Accept {
	ConfigKeyNot ragnaFieldScoutOff 1
	JobID 0
	BaseLevel <= 25
	QuestInactive 60202
	IsInMapAndCoordinate prt_fild08
	timeout 70
	exclusive 1
	call {
		[
			log Ragna Field Scout accept hunt 20 kills
			do conf attackAuto 0
			do conf route_randomWalk 0
			do move 160 368
			pause 22
			do talknpc 160 370 c c r0
			pause 14
			do conf lockMap prt_fild08
			do conf lockMap_x none
			do conf lockMap_y none
			do conf attackRouteMaxPathDistance 50
			do conf attackAuto 2
			do conf route_randomWalk 1
			do conf route_randomWalk_maxRouteTime 60
			log Ragna Field Scout active hunting prt_fild08
		]
	}
}
