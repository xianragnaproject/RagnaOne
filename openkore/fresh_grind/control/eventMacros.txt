###############################################################################
# Sell @ 40% OW via built-in sellAuto (items_control must autoSell junk)
# Standpoint 130,72 is walkable; NPC talk uses distance (do NOT stand on 126,74)
###############################################################################
automacro Grind_SellLoot {
	exclusive 1
	timeout 60
	priority 2
	ConfigKeyNot grindAfk 1
	ConfigKeyNot grindSelling 1
	ConfigKeyNot grindJobbing 1
	ConfigKeyNot grindDone25 1
	Eval (Misc::percent_weight($char) >= 40) && !(AI::inQueue("sellAuto") || AI::inQueue("buyAuto") || AI::is("sellAuto","buyAuto","NPC"))
	call grind_sell_loot
}

macro grind_sell_loot {
	[
		log Grind: OW>=40% - sell at Prontera tool dealer
		do conf grindSelling 1
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sellAuto 1
		do conf sellAuto_standpoint prt_in 130 72
		do conf sellAuto_npc prt_in 126 76
		do conf sellAuto_distance 10
		do conf sellAuto_maxDistance 14
		do move prt_in 130 72
		pause 35
		do eval AI::queue("sellAuto")
		pause 35
		do move 135 71
		pause 6
		do move prontera 160 70
		pause 12
		do eval Grind_ResumeHuntLock()
		do conf grindSelling 0
		log Grind: sell done - resume hunt
	]
}

	]
}

sub Grind_TalkJobMaster {
	my $job = lc($config{grindTargetJob} || 'Thief');
	my %menu = (
		swordman => 'r0',
		swordsman => 'r0',
		knight => 'r0',
		magician => 'r1',
		mage => 'r1',
		wizard => 'r1',
		archer => 'r2',
		hunter => 'r2',
		acolyte => 'r3',
		priest => 'r3',
		merchant => 'r4',
		blacksmith => 'r4',
		thief => 'r5',
		assassin => 'r5',
	);
	my $r = $menu{$job} || 'r5';
	Log::message("[Grind] Job Master talk for '$job' ($r) at prontera 155,180\n", "success");
	Commands::run("talknpc 155 180 c c c $r c r0");
	return 1;
}

# Enable only this job's gear buyAuto blocks (potions/wings stay as-is)
sub Grind_EnableGearBuy {
	my $jid = $char->{jobID};
	my %want = (
		1 => { 'Sword [3]' => 1, 'Guard' => 1, 'Cotton Shirt' => 1 },
		2 => { 'Rod [3]' => 1, 'Cotton Shirt' => 1 },
		3 => { 'Bow [3]' => 1, 'Arrow' => 1, 'Cotton Shirt' => 1 },
		4 => { 'Club' => 1, 'Cotton Shirt' => 1 },
		5 => { 'Axe [3]' => 1, 'Cotton Shirt' => 1 },
		6 => { 'Knife [3]' => 1, 'Cotton Shirt' => 1 },
	);
	my $map = $want{$jid} || {};
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || '';
		next if $item =~ /Potion|Wing|Magnifier/i;
		my $disabled = $map->{$item} ? 0 : 1;
		Commands::run("conf buyAuto_${i}_disabled $disabled");
	}
	Log::message("[Grind] Enabled gear buyAuto for jobID=$jid\n", "success");
	return 1;
}

sub Grind_StartHunt {
	Commands::run("conf lockMap prt_fild08");
	Commands::run("conf lockMap_x none");
	Commands::run("conf attackAuto 2");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf runFromTarget 0");
	Commands::run("conf teleportAuto_hp 0");
	Commands::run("conf sitAuto_hp_lower 35");
	Log::message("[Grind] Hunting prt_fild08 until base 25\n", "success");
	return 1;
}

sub Grind_ResumeHuntLock {
	if ($char && $char->{jobID} == 0 && $char->{lv} >= 15 && $char->{lv_job} >= 10) {
		# Novice ready for job change - stay put, job macros take over
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	if ($config{grindJob1st} && !$config{grindGearDone}) {
		# Still shopping for class gear
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Grind_StartHunt();
	return 1;
}
