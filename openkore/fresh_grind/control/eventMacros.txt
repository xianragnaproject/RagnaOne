macro grind_fountain_afk {
	$secs = &rand(300, 600)
	[
		log Grind: AFK break - portal then fountain for $secs sec
		do conf grindAfk 1
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sitAuto_idle 0
	]
	call grind_to_prontera
	[
		do eval Grind_Phase1SitIfInTown()
		pause 70
	]
	call grind_afk_face_sit
	[
		pause $secs
		do stand
		log Grind: AFK done - resume hunt
		do conf sitAuto_idle 1
		do eval Grind_ResumeHuntLock()
		do conf grindAfk 0
	]
}

	return 0 unless $field;

}

# Same-map walk to walkable approach tile beside north portal.
# Portal src is prt_fild08 170,378 (not a pathfind dest).
sub Grind_ApproachFieldPortal {
	return 1 if Grind_IsInPronteraTown();
	return 1 unless $field && $field->baseName eq 'prt_fild08';
	my $x = $char->{pos}{x} // 0;
	my $y = $char->{pos}{y} // 0;
	return 1 if abs($x - 170) <= 2 && abs($y - 375) <= 3;
	Commands::run("move 170 375");
	return 1;
}

# From approach tile, enter portal then land at Prontera south gate.
# One action per call — grind_to_prontera invokes this several times with pauses.
sub Grind_EnterFieldPortal {
	return 1 if Grind_IsInPronteraTown();
	return 1 unless $field && $field->baseName eq 'prt_fild08';
	my $x = $char->{pos}{x} // 0;
	my $y = $char->{pos}{y} // 0;
	unless (abs($x - 170) <= 6 && abs($y - 375) <= 10) {
		Commands::run("stand") if $char->{sitting};
		Commands::run("move 170 375");
		return 1;
	}
	Commands::run("stand") if $char->{sitting};
	# Adjacent to portal: step north into warp (170,378), else short MapRoute
	if ($y < 377) {
		Commands::run("move 170 377");
	} elsif ($y < 378) {
		# Last step onto portal cell — may log "not walkable" but still warps
		Commands::run("north");
	} else {
		Commands::run("move prontera 156 30");
	}
	return 1;
}

# Body dirs: 0=N 1=NW 2=W 3=SW 4=S 5=SE 6=E 7=NE
# Walking one tile is the only reliable way other players see facing on this server.
# (`do sit` / Task::SitStand ignores sitAuto_look; look packets while idle often no-op.)
sub Grind_StepFace {
	return 0 unless $char && $field && $field->baseName eq 'prontera';
	Commands::run("stand") if $char->{sitting};
	AI::clear("move", "route", "mapRoute", "sitting", "standing", "look");

	my @dx = (0, -1, -1, -1, 0, 1, 1, 1);
	my @dy = (1, 1, 0, -1, -1, -1, 0, 1);
	my @dir = qw(N NW W SW S SE E NE);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;

	my @cands;
	for my $d (0 .. 7) {
		my ($tx, $ty) = ($x + $dx[$d], $y + $dy[$d]);
		next unless $ty >= 170 && abs($tx - 156) <= 22;
		# Prefer tiles we know are plaza; isWalkable is unreliable on water here
		push @cands, [$d, $tx, $ty];
	}
	unless (@cands) {
		for my $d (0 .. 7) {
			push @cands, [$d, $x + $dx[$d], $y + $dy[$d]];
		}
	}
	# Shuffle
	for (my $i = @cands - 1; $i > 0; $i--) {
		my $j = int(rand($i + 1));
		@cands[$i, $j] = @cands[$j, $i];
	}
	my ($body, $tx, $ty) = @{$cands[0]};
	Commands::run("conf sitAuto_look $body");
	Commands::run("conf sitAuto_look_delay 0");
	Commands::run("move $tx $ty");
	Log::message("[Grind] AFK step face=$body ($dir[$body]) -> $tx,$ty\n", "success");
	return $body;
}

	return 1;
}

# Pick a random tile on the Prontera fountain plaza (south/side of the water).
# Do NOT trust isWalkable alone on this server build — it marks water as walkable.
sub Grind_RandomFountainTile {
	my @spots = (
		# South plaza (facing fountain) — standard AFK strip
		[155, 185], [150, 185], [160, 185], [148, 185], [162, 185],
		[155, 180], [152, 182], [158, 182], [150, 182], [160, 182],
		[155, 188], [152, 188], [158, 188], [148, 188], [162, 188],
		[155, 190], [150, 190], [160, 190], [152, 190], [158, 190],
		# East / west apron
		[145, 192], [146, 188], [166, 192], [167, 188],
		[145, 185], [167, 185],
	);
	my @ok;
	if ($field && $field->baseName eq 'prontera') {
		for my $s (@spots) {
			# Keep only tiles OpenKore will actually accept for move
			next unless $field->isWalkable($s->[0], $s->[1]);
			push @ok, $s;
		}
	}
	# If filter empty OR filter only returned water-false-positives, use south plaza raw
	@ok = (
		[155, 185], [150, 185], [160, 185], [152, 182], [158, 182],
		[155, 180], [148, 185], [162, 185], [155, 188],
	) unless @ok;
	my $s = $ok[int(rand(scalar @ok))];
	return ($s->[0], $s->[1]);
}

			list => 'Basic Skill 9, Sword Mastery 10, Bash 10, Magnum Break 10, Provoke 5, Endure 5, Increase HP Recovery 5',
			skills => ['Bash', 'Magnum Break', 'Provoke'],
		},
		2 => {
			name => 'Magician',
			list => 'Basic Skill 9, Increase SP Recovery 8, Sight 1, Fire Bolt 10, Cold Bolt 5, Lightning Bolt 5, Napalm Beat 5, Fire Ball 5',
			skills => ['Fire Bolt', 'Cold Bolt', 'Lightning Bolt', 'Napalm Beat'],
		},
		3 => {
			name => 'Archer',
			list => "Basic Skill 9, Owl's Eye 10, Vulture's Eye 10, Double Strafe 10, Improve Concentration 5, Arrow Shower 10, Charge Arrow 1",
			skills => ['Double Strafe', 'Arrow Shower', 'Charge Arrow', 'Improve Concentration'],
		},
		4 => {
			name => 'Acolyte',
			list => 'Basic Skill 9, Heal 10, Increase Agility 10, Blessing 10, Angelus 5, Cure 1, Divine Protection 5, Demon Bane 3, Holy Light 1, Ruwach 1',
			skills => ['Holy Light', 'Heal', 'Angelus', 'Blessing', 'Increase Agility'],
		},
		5 => {
			name => 'Merchant',
			list => 'Basic Skill 9, Enlarge Weight Limit 10, Mammonite 10, Discount 5, Overcharge 5, Pushcart 10, Item Appraisal 1, Vending 1',
			skills => ['Mammonite'],
		},
		6 => {
			name => 'Thief',
			list => 'Basic Skill 9, Double Attack 10, Improve Dodge 10, Envenom 5, Steal 4, Hiding 2, Detoxify 1',
			skills => ['Envenom'],
		},
	);
	my $spec = $by_id{$jid};
	unless ($spec) {
		Log::message("[Grind] ApplyClassSkills: unsupported jobID=$jid\n", "system");
		return 0;
	}
	my %want = map { $_ => 1 } @{$spec->{skills}};
	Commands::run("conf skillsAddAuto 1");
	Commands::run("conf skillsAddAuto_list $spec->{list}");
	Commands::run("conf -f grindTargetJob $spec->{name}") if !$config{grindTargetJob} || $config{grindTargetJob} =~ /^(random|none|auto)?$/i;

	for my $prefix (qw(attackSkillSlot useSelf_skill partySkill)) {
		for (my $i = 0; exists $config{"${prefix}_$i"} || ($i == 0 && exists $config{$prefix}); $i++) {
			my $name = $config{"${prefix}_$i"} || '';
			next unless $name =~ /\S/;
			next if $name =~ /^\{/;
			my $disabled = $want{$name} ? 0 : 1;
			if ($prefix eq 'partySkill' && $name eq 'Heal' && $jid == 4) {
				$disabled = 0;
			}
			Commands::run("conf ${prefix}_${i}_disabled $disabled");
		}
	}
	if ($jid == 5) {
		Grind_SetMammoniteDisabled(($char->{zeny} // 0) < 1500 ? 1 : 0);
	}
	Log::message("[Grind] Class skills enabled for $spec->{name} (jobID=$jid)\n", "success");
	return 1;
}

sub Grind_SetMammoniteDisabled {
	my ($disabled) = @_;
	$disabled = 1 unless defined $disabled;
	for (my $i = 0; exists $config{"attackSkillSlot_$i"} || ($i == 0 && exists $config{attackSkillSlot}); $i++) {
		my $name = $config{"attackSkillSlot_$i"} || '';
		next unless $name eq 'Mammonite';
		Commands::run("conf attackSkillSlot_${i}_disabled $disabled");
		Commands::run("conf -f attackSkillSlot_${i}_zeny > 1500") if !$disabled;
		last;
	}
	return 1;
}

sub Grind_DisableBuyItem {
	my ($name) = @_;
	return 0 unless defined $name && $name ne '';
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || '';
		next unless $item eq $name;
		Commands::run("conf buyAuto_${i}_disabled 1");
	}
	return 1;
}

sub Grind_EnableBuyItem {
	my ($name) = @_;
	return 0 unless defined $name && $name ne '';
	my $changed = 0;
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || '';
		next unless $item eq $name;
		next if defined $config{"buyAuto_${i}_disabled"} && $config{"buyAuto_${i}_disabled"} eq '0';
		Commands::run("conf buyAuto_${i}_disabled 0");
		$changed = 1;
	}
	Log::message("[Grind] Enabled buyAuto for $name\n", "success") if $changed;
	return 1;
}

sub Grind_StartHunt {
	return 1 if $config{grindAfk};
	# Phase1 park: no hunt unless Phase2 Swordman path is active
	if ($config{grindPhase1Done} && !$config{grindPhase2}) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Commands::run("conf follow 0");
	Commands::run("conf followTarget");
	Commands::run("conf grindPartyMode 0");
	my $map = $config{grindHuntMap};
	if ($config{grindPhase2} && $config{grindP2Gear}) {
		$map = 'pay_fild08';
		Commands::run("conf -f grindHuntMap $map");
	} elsif (!$map || $map eq '' || $map eq 'none') {
		$map = 'prt_fild08';
		Commands::run("conf -f grindHuntMap $map");
	}
	Commands::run("conf lockMap $map");
	Commands::run("conf lockMap_x none");
	Commands::run("conf attackAuto 2");
	Commands::run("conf attackAuto_inLockOnly 1");
	Commands::run("conf attackAuto_followTarget 0");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf runFromTarget 0");
	Commands::run("conf sellAuto 0");
	Commands::run("conf teleportAuto_hp 0");
	Commands::run("conf sitAuto_hp_lower 35");
	Log::message("[Grind] Solo hunting $map\n", "success");
	return 1;
}

sub Grind_ResumeHuntLock {
	if ($config{grindAfk}) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	if ($config{grindPhase1Done} && !$config{grindPhase2}) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	if ($char && $char->{jobID} == 0 && $char->{lv_job} >= 10) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Grind_StartHunt();
	return 1;
}

# Remap sell + potion buy + gear buy NPCs to Payon shops
sub Grind_ApplyPayonShops {
	Commands::run("conf sellAuto_npc payon_in01 5 49");
	Commands::run("conf sellAuto_standpoint payon_in01 7 49");
	Commands::run("conf sellAuto_distance 10");
	Commands::run("conf sellAuto_maxDistance 14");
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || "";
		my ($npc, $stand);
		if ($item =~ /^(Red Potion|Fly Wing|Butterfly Wing|Arrow)$/i) {
			$npc = "payon_in01 5 49";
			$stand = "payon_in01 7 49";
		} elsif ($item =~ /^(Shoes|Wooden Mail|Manteau|Muffler|Cotton Shirt|Guard)$/i) {
			$npc = "payon_in01 7 119";
			$stand = "payon_in01 10 119";
		} elsif ($item =~ /^(Blade|Sword|Knife|Rod|Bow|Axe)/i) {
			$npc = "payon_in01 15 119";
			$stand = "payon_in01 15 117";
		} else {
			next;
		}
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand");
	}
	Log::message("[Grind] Payon shops applied (tool/armor/weapon)\n", "success");
	return 1;
}

sub Grind_SetBuyNpc {
	my ($name, $npc, $stand) = @_;
	return 0 unless $name && $npc;
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		next unless ($config{"buyAuto_$i"} || "") eq $name;
		Commands::run("conf buyAuto_${i}_npc $npc");
		Commands::run("conf buyAuto_${i}_standpoint $stand") if $stand;
		return 1;
	}
	return 0;
}

sub Grind_Phase2Equip {
	my $inv = $char && $char->inventory;
	for my $name ("Blade", "Guard", "Wooden Mail", "Muffler", "Shoes") {
		next unless $inv && $inv->getByName($name);
		Commands::run("eq $name");
	}
	# Enable only the Muffler swordsman set; disable competing Swordsman equipAuto blocks
	for (my $i = 0; exists $config{"equipAuto_$i"}; $i++) {
		my $job = $config{"equipAuto_${i}_isJob"} || "";
		next unless $job =~ /Swordsman/i;
		my $robe = $config{"equipAuto_${i}_robe"} || "";
		my $armor = $config{"equipAuto_${i}_armor"} || "";
		if ($robe eq "Muffler" && $armor eq "Wooden Mail") {
			Commands::run("conf equipAuto_${i}_disabled 0");
		} else {
			Commands::run("conf equipAuto_${i}_disabled 1");
		}
	}
	Log::message("[Grind] PHASE2 equip Blade/Guard/Wooden Mail/Muffler/Shoes\n", "success");
	return 1;
}

# True if any core Payon armor piece is missing (inventory includes equipped).
sub Grind_Phase2NeedsArmor {
	return 0 unless $char && $char->inventory;
	my $inv = $char->inventory;
	return 1 unless $inv->getByName("Shoes");
	return 1 unless $inv->getByName("Wooden Mail");
	return 1 unless ($inv->getByName("Muffler") || $inv->getByName("Manteau"));
	return 0;
}

# payon_in01 15,119 weapon shop: idx 4 = Blade (1107)
# Sequence: b = open buy, bN,A = buy store index N amount A (MUST be bN,A not "b N,A")
sub Grind_Phase2TalkBuyBlade {
	return 1 if $char && $char->inventory && $char->inventory->getByName("Blade");
	Commands::run('talknpc 15 119 b b4,1 e');
	Log::message("[Grind] PHASE2 talknpc buy Blade: b b4,1 e\n", "success");
	return 1;
}

# payon_in01 7,119 armor shop indices (from npc_shops.txt):
# 1=Shoes(2403) 4=Muffler(2503) 13=Wooden Mail(2328)
# Prefer one-item talknpc from macros; this helper buys only still-missing pieces.
sub Grind_Phase2TalkBuyArmor {
	my $inv = $char && $char->inventory;
	my @seqs;
	push @seqs, 'b b1,1 e' unless $inv && $inv->getByName("Shoes");
	push @seqs, 'b b4,1 e' unless $inv && ($inv->getByName("Muffler") || $inv->getByName("Manteau"));
	push @seqs, 'b b13,1 e' unless $inv && $inv->getByName("Wooden Mail");
	unless (@seqs) {
		Log::message("[Grind] PHASE2 armor already owned\n", "success");
		return 1;
	}
	# Queue first missing piece only — macros call this repeatedly / buy one-by-one
	my $seq = $seqs[0];
	Commands::run("talknpc 7 119 $seq");
	Log::message("[Grind] PHASE2 talknpc buy armor: $seq\n", "success");
	return 1;
}

# After armor talknpc: resume hunt only if Shoes + Wooden Mail + (Muffler|Manteau) owned
sub Grind_Phase2FinishArmorBuy {
	if (Grind_Phase2NeedsArmor()) {
		Log::message("[Grind] PHASE2 armor STILL missing after buy — stay in Payon to retry\n", "success");
		Commands::run("conf lockMap none");
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		return 0;
	}
	Log::message("[Grind] PHASE2 armor OK — resume pay_fild08\n", "success");
	Commands::run("conf -f grindHuntMap pay_fild08");
	Commands::run("conf lockMap pay_fild08");
	Commands::run("conf attackAuto 2");
	Commands::run("conf route_randomWalk 1");
	Commands::run("move payon 267 89");
	return 1;
}

# After Alberta armor buy: leave buying flag; return automacro warps to Payon
sub Grind_Phase2FinishArmorBuyFromAlberta {
	if (Grind_Phase2NeedsArmor()) {
		Log::message("[Grind] PHASE2 Alberta armor STILL missing — retry buy\n", "success");
		Commands::run("conf grindP2Buying 1");
		Commands::run("conf lockMap none");
		Commands::run("conf attackAuto 0");
		return 0;
	}
	Log::message("[Grind] PHASE2 Alberta armor OK — ready to return to Payon\n", "success");
	Commands::run("conf attackAuto 0");
	Commands::run("conf route_randomWalk 0");
	return 1;
}
