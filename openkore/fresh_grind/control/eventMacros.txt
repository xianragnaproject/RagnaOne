	[
		log Grind: OW>=40% - sell at Prontera tool dealer

		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sellAuto 1

		do conf sellAuto_npc prt_in 126 76
		do conf sellAuto_distance 10
		do conf sellAuto_maxDistance 14
		do move prt_in 130 72

		do eval AI::queue("sellAuto")

		do move 135 71
		pause 6
		do move prontera 160 70
		pause 12
		do eval Grind_ResumeHuntLock()
		do conf grindSelling 0
		log Grind: sell done - resume hunt
	]
}

# Enable only this job's gear buyAuto + equipAuto blocks (potions/wings stay as-is)
sub Grind_EnableGearBuy {
	my $jid = $char->{jobID};
	my %want = (
		1 => { 'Sword [3]' => 1, 'Guard' => 1, 'Cotton Shirt' => 1 },
		2 => { 'Rod [3]' => 1, 'Cotton Shirt' => 1 },
		3 => { 'Bow [3]' => 1, 'Arrow' => 1, 'Cotton Shirt' => 1 },
		4 => { 'Club' => 1, 'Cotton Shirt' => 1 },
		5 => { 'Axe [3]' => 1, 'Cotton Shirt' => 1 },
		6 => { 'Knife [3]' => 1, 'Cotton Shirt' => 1 },
	);
	my %wantEquip = (
		1 => 'Sword [3]',
		2 => 'Rod [3]',
		3 => 'Bow [3]',
		4 => 'Club',
		5 => 'Axe [3]',
		6 => 'Knife [3]',
	);
	my $map = $want{$jid} || {};
	for (my $i = 0; exists $config{"buyAuto_$i"}; $i++) {
		my $item = $config{"buyAuto_$i"} || '';
		next if $item =~ /Potion|Wing|Magnifier/i;
		my $disabled = $map->{$item} ? 0 : 1;
		Commands::run("conf buyAuto_${i}_disabled $disabled");
	}
	# JobID is ignored by equipAuto — gate with disabled + isJob in config
	my $wantRh = $wantEquip{$jid} || '';
	for (my $i = 0; exists $config{"equipAuto_$i"} || defined $config{"equipAuto_${i}_rightHand"}; $i++) {
		my $rh = $config{"equipAuto_${i}_rightHand"} || '';
		next unless $rh =~ /\S/;
		my $disabled = ($wantRh ne '' && $rh eq $wantRh) ? 0 : 1;
		Commands::run("conf equipAuto_${i}_disabled $disabled");
	}
	Log::message("[Grind] Enabled gear buy/equip for jobID=$jid ($wantRh)\n", "success");
	return 1;
}

sub Grind_FollowerFarFromLeader {
	return 0 unless $config{follow} && $config{followTarget} && $char && $char->{party};
	my $id = main::findPartyUserID($config{followTarget});
	return 0 if !defined $id || $id eq '';
	my $u = $char->{party}{users}{$id};
	return 0 unless $u;
	my $mmap = '';
	if ($u->{map}) {
		($mmap) = $u->{map} =~ /([\s\S]*)\.gat/;
		$mmap = $u->{map} unless defined $mmap && $mmap ne '';
		$mmap =~ s/\.gat$//;
	}
	my $fmap = ($field && $field->baseName) ? $field->baseName : '';
	return 1 if $mmap && $fmap && $mmap ne $fmap;
	my $mx = $u->{pos}{x};
	my $my = $u->{pos}{y};
	return 0 unless defined $mx && defined $my && $mx ne '' && $my ne '';
	my $cx = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $cy = $char->{pos_to}{y} // $char->{pos}{y} // 0;
	my $dist = abs($cx - $mx) + abs($cy - $my);
	return $dist > 22;
}

sub Grind_StartHunt {
	return 1 if $config{grindBuyingArrows};
	# Party followers stick to the leader instead of locking the field
	if ($config{follow} && $config{followTarget}) {
		Grind_ApplyFollowerFollow();
		Log::message("[Grind] Following ".$config{followTarget}." (party support)\n", "success");
		return 1;
	}
	my $map = $config{grindHuntMap};
	if (!$map || $map eq '' || $map eq 'none') {
		$map = ($char && $char->{lv} >= 25) ? 'pay_fild03' : 'prt_fild08';
		Commands::run("conf -f grindHuntMap $map");
	}
	Commands::run("conf lockMap $map");
	Commands::run("conf lockMap_x none");
	Commands::run("conf attackAuto 2");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf runFromTarget 0");
	Commands::run("conf teleportAuto_hp 0");
	Commands::run("conf sitAuto_hp_lower 35");
	if ($map eq 'pay_fild03') {
		Commands::run("conf attackAuto_routeToLock 0");
		Commands::run("conf attackAuto_outOfLock 0");
		Commands::run("conf teleportAuto_deadly 1");
	}
	Log::message("[Grind] Hunting $map (party leader)\n", "success");
	return 1;
}

sub Grind_ResumeHuntLock {
	if ($char && $char->{jobID} == 0 && $char->{lv} >= 15 && $char->{lv_job} >= 10) {
		# Novice ready for job change - stay put, job macros take over
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	if ($config{grindJob1st} && !$config{grindGearDone}) {
		# Still shopping for class gear
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Grind_StartHunt();
	return 1;
}
