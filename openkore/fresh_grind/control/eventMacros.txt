automacro Grind_UnstickPronteraSouthPos {
	disabled 1
	exclusive 1
	timeout 60
	priority 1
	InMap prontera
	ConfigKeyNot grindAfk 1
	ConfigKeyNot grindJobbing 1
	ConfigKeyNot grindSelling 1
	ConfigKeyNot grindUnsticking 1
	ConfigKey lockMap prt_fild08
	Eval abs(($char->{pos}{x}//0)-155) <= 8 && abs(($char->{pos}{y}//0)-183) <= 12
	call grind_unstick_prt_south
}

macro grind_unstick_prt_south {
	log Grind: unstick Prontera spawn - one walk to south gate
	do eval $config{grindUnsticking} = 1; $config{lockMap} = ''; $config{route_randomWalk} = 0; $config{attackAuto} = 0; 1
	do stand
	do eval AI::clear("move", "route", "mapRoute")
	do move 156 22
	pause 90
	do conf lockMap prt_fild08
	do conf attackAuto 2
	do conf route_randomWalk 1
	do conf -f grindHuntMap prt_fild08
	do eval $config{grindUnsticking} = 0; 1
}

# Body dirs: 0=N 1=NW 2=W 3=SW 4=S 5=SE 6=E 7=NE
# Walking one tile is the only reliable way other players see facing on this server.
# (`do sit` / Task::SitStand ignores sitAuto_look; look packets while idle often no-op.)
sub Grind_StepFace {
	return 0 unless $char && $field && $field->baseName eq 'prontera';
	Commands::run("stand") if $char->{sitting};
	AI::clear("move", "route", "mapRoute", "sitting", "standing", "look");

	my @dx = (0, -1, -1, -1, 0, 1, 1, 1);
	my @dy = (1, 1, 0, -1, -1, -1, 0, 1);
	my @dir = qw(N NW W SW S SE E NE);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;

	my @cands;
	for my $d (0 .. 7) {
		my ($tx, $ty) = ($x + $dx[$d], $y + $dy[$d]);
		next unless $ty >= 170 && abs($tx - 156) <= 22;
		# Prefer tiles we know are plaza; isWalkable is unreliable on water here
		push @cands, [$d, $tx, $ty];
	}
	unless (@cands) {
		for my $d (0 .. 7) {
			push @cands, [$d, $x + $dx[$d], $y + $dy[$d]];
		}
	}
	# Shuffle
	for (my $i = @cands - 1; $i > 0; $i--) {
		my $j = int(rand($i + 1));
		@cands[$i, $j] = @cands[$j, $i];
	}
	my ($body, $tx, $ty) = @{$cands[0]};
	Commands::run("conf sitAuto_look $body");
	Commands::run("conf sitAuto_look_delay 0");
	Commands::run("move $tx $ty");
	Log::message("[Grind] AFK step face=$body ($dir[$body]) -> $tx,$ty\n", "success");
	return $body;
}

sub Grind_AfkSit {
	return 1 if $char && $char->{sitting};
	# Prevent processSitAuto from standing us immediately (full HP in town)
	$ai_v{sitAuto_forcedBySitCommand} = 1;
	$ai_v{sitAuto_forceStop} = 0;
	Commands::run("sit");
	return 1;
}

}

	Commands::run("stand") if $char && $char->{sitting};

}

# Pick a random tile on the Prontera fountain plaza (south/side of the water).
# Do NOT trust isWalkable alone on this server build — it marks water as walkable.
sub Grind_RandomFountainTile {
	my @spots = (
		# South plaza (facing fountain) — standard AFK strip
		[155, 185], [150, 185], [160, 185], [148, 185], [162, 185],
		[155, 180], [152, 182], [158, 182], [150, 182], [160, 182],
		[155, 188], [152, 188], [158, 188], [148, 188], [162, 188],
		[155, 190], [150, 190], [160, 190], [152, 190], [158, 190],
		# East / west apron
		[145, 192], [146, 188], [166, 192], [167, 188],
		[145, 185], [167, 185],
	);
	my @ok;
	if ($field && $field->baseName eq 'prontera') {
		for my $s (@spots) {
			# Keep only tiles OpenKore will actually accept for move
			next unless $field->isWalkable($s->[0], $s->[1]);
			push @ok, $s;
		}
	}
	# If filter empty OR filter only returned water-false-positives, use south plaza raw
	@ok = (
		[155, 185], [150, 185], [160, 185], [152, 182], [158, 182],
		[155, 180], [148, 185], [162, 185], [155, 188],
	) unless @ok;
	my $s = $ok[int(rand(scalar @ok))];
	return ($s->[0], $s->[1]);
}


# Dump free skill points into raisable skills (Basic Skill first) before Job Master.
sub Grind_SpendSkillPoints {
	return 0 unless $char && $messageSender;
	my $before = $char->{points_skill} // 0;
	return 1 if $before < 1;
	my $sent = 0;
	my @handles = @skillsID;
	@handles = sort {
		(($a eq 'NV_BASIC') ? 0 : 1) <=> (($b eq 'NV_BASIC') ? 0 : 1)
	} @handles;
	for my $handle (@handles) {
		last if ($char->{points_skill} // 0) < 1;
		next unless $char->{skills}{$handle};
		next unless ($char->{skills}{$handle}{up} // 0);
		my $skill = Skill->new(handle => $handle);
		next unless $skill && $skill->getIDN;
		while (($char->{points_skill} // 0) > 0 && ($char->{skills}{$handle}{up} // 0)) {
			$messageSender->sendAddSkillPoint($skill->getIDN);
			$sent++;
			# Optimistic local update so we do not flood before ACK
			$char->{points_skill}--;
			$char->{skills}{$handle}{lv} = ($char->{skills}{$handle}{lv} // 0) + 1;
			last if $sent >= 15;
		}
	}
	Log::message("[Grind] Spent skill points (sent=$sent, had=$before)\n", "success") if $sent;
	return 1;
}

# Equip free starter gear granted on 1st job change (weapon + cotton shirt + guard)
sub Grind_EquipJobChangeGear {
	return 0 unless $char;
	my $jid = $char->{jobID} // 0;
	my %weapons = (
		1 => ['Sword', 'Sword [3]', 'Blade', 'Knife', 'Knife [3]'],
		2 => ['Rod', 'Rod [3]', 'Staff', 'Knife', 'Knife [3]'],
		3 => ['Bow', 'Bow [3]', 'Arrow'],
		4 => ['Club', 'Club [3]', 'Mace', 'Rod', 'Rod [3]'],
		5 => ['Axe', 'Axe [3]', 'Axe [4]', 'Knife', 'Knife [3]'],
		6 => ['Main Gauche', 'Main Gauche [3]', 'Main Gauche', 'Main Gauche [3]', 'Knife', 'Knife [3]', 'Dagger'],
	);
	my @armor = ('Cotton Shirt', 'Cotton Shirt [1]', 'Cotton Shirt', 'Novice False Eggshell', 'Novice Guard', 'Guard', 'Sandals', 'Hood', 'Cap');
	my @try = @{ $weapons{$jid} || ['Knife', 'Knife [3]', 'Sword'] };
	push @try, @armor;
	my $equipped = 0;
	for my $name (@try) {
		next if $name eq 'Arrow'; # arrows use eq arrow
		my $item = Actor::Item::get($name, undef, 1);
		next unless $item;
		next if $item->{equipped};
		eval { $item->equip(); };
		if ($item->{equipped} || !$@) {
			Commands::run("eq $name");
			$equipped++;
			Log::message("[Grind] Equipped job-change gear: $name\n", "success");
		}
	}
	# Archers need arrows equipped into arrow slot
	if ($jid == 3) {
		for my $an ('Arrow', 'Iron Arrow', 'Fire Arrow') {
			my $a = Actor::Item::get($an, undef, 1);
			next unless $a;
			Commands::run("eq $an");
			Log::message("[Grind] Equipped arrows: $an\n", "success");
			last;
		}
	}
	Commands::run("conf -f grindJobGearEquipped 1") if $equipped;
	return 1;
}

sub Party12_StickToLeader {
	return 1 unless (($config{partyRole} || '') eq 'member');

	# Unfinished Novice: always solo (even if sell/afk flags stuck on)
	if ($char && (($char->{jobID} // 0) == 0) && (($config{grindJob1st} || '') ne '1')) {
		Party12_NoviceSoloUntilJob();
		return 1;
	}

	# Let job-change / sell macros own movement
	return 1 if (($config{grindJobbing} || '') eq '1' || ($config{grindSelling} || '') eq '1' || ($config{grindAfk} || '') eq '1');

	Party12_ApplyMemberFollow();
	my $lead = $config{followTarget} || $config{partyLeaderName} || '';
	return 1 if $lead eq '';
	my $near = 0;
	my $max = ($config{followDistanceMax} || 4) + 3;
	if ($playersList) {
		for my $p (@$playersList) {
			next unless $p;
			my $n = eval { $p->name };
			next unless defined $n && $n eq $lead;
			my $d = blockDistance($char->{pos_to}, $p->{pos_to});
			$near = 1 if defined $d && $d <= $max;
			last;
		}
	}
	# Far from master: attackAuto -1 (0 still retaliates when Porings slap you
	# while walking — that stalls follow + blocks EXP Even Share).
	# Near master: assist-only (attackAuto 1) so they help kill his targets.
	if (!$near) {
		Commands::run("conf attackAuto -1") if (($config{attackAuto} // '') ne '-1');
		if ($char) { eval { $char->sendAttackStop }; }
		AI::clear("attack");
	} else {
		Commands::run("conf attackAuto 1") if (($config{attackAuto} // '') ne '1');
		Commands::run("conf attackAuto_party 1");
		Commands::run("conf attackAuto_followTarget 1");
		Commands::run("conf attackChangeTarget 1");
		# Near master: drop stuck route/move so assist can fire
		if (AI::action() eq "move" || (AI::action() eq "route" && !(AI::args() && AI::args()->{isFollow}))) {
			AI::clear("move", "route");
		}
	}
	ai_follow($lead) if defined &ai_follow;
	return 1;
}

sub Grind_Phase2Equip {
	my $inv = $char && $char->inventory;
	my $weapon = Grind_Phase2WeaponName();
	for my $name ($weapon, "Guard", "Wooden Mail", "Muffler", "Shoes") {
		next unless $inv;
		my $item = $inv->getByName($name);
		unless ($item) {
			for my $it (@{$inv->getItems}) {
				my $n = $it->name // next;
				if ($n eq $name || $n =~ /^\Q$name\E(?:\s*\[|$)/) {
					$item = $it;
					last;
				}
			}
		}
		next unless $item;
		next if $item->{equipped};
		Commands::run("eq " . $item->name);
	}
	# Enable full Phase2 set; disable competing same-job equipAuto blocks
	for (my $i = 0; exists $config{"equipAuto_$i"}; $i++) {
		my $job = $config{"equipAuto_${i}_isJob"} || "";
		my $robe = $config{"equipAuto_${i}_robe"} || "";
		my $armor = $config{"equipAuto_${i}_armor"} || "";
		my $rh = $config{"equipAuto_${i}_rightHand"} || "";
		if (Grind_Phase2IsSwordman() && $job =~ /Swordsman/i) {
			if ($robe eq "Muffler" && $armor eq "Wooden Mail") {
				Commands::run("conf equipAuto_${i}_disabled 0");
			} else {
				Commands::run("conf equipAuto_${i}_disabled 1");
			}
		} elsif (Grind_Phase2IsMerchant() && $job =~ /Merchant/i) {
			if ($rh =~ /^Axe$/i && $armor eq "Wooden Mail" && $robe eq "Muffler") {
				Commands::run("conf equipAuto_${i}_disabled 0");
			} elsif ($rh =~ /^Axe/i && $armor eq "Cotton Shirt") {
				# Keep starter Axe [3] set only until full armor is owned
				my $disable = (Grind_Phase2HasItem("Wooden Mail") && Grind_Phase2HasRobe()) ? 1 : 0;
				Commands::run("conf equipAuto_${i}_disabled $disable");
			} else {
				Commands::run("conf equipAuto_${i}_disabled 1");
			}
		}
	}
	Log::message("[Grind] PHASE2 equip $weapon/Guard/Wooden Mail/Muffler/Shoes\n", "success");
	return 1;
}

		return 1 if $n eq $name;
		return 1 if $n =~ /^\Q$name\E(?:\s*\[|$)/;
	}
	return 0;
}

# Periodic console audit of Phase2 gear ownership
sub Grind_Phase2LogGear {
	return 0 unless $char && $char->inventory;
	my $inv = $char->inventory;
	my @parts;
	for my $item (@{$inv->getItems}) {
		my $n = $item->name // next;
		next unless $n =~ /Blade|Axe|Guard|Shoes|Muffler|Manteau|Wooden Mail|Knife|Sword|Cotton Shirt/;
		push @parts, $n . ($item->{equipped} ? "*" : "");
	}
	my $z = $char->{zeny} // 0;
	my $map = $field ? $field->baseName : "?";
	Log::message("[Grind] PHASE2 GEAR ".$char->{name}." z=$z map=$map [".join(",", @parts)."]\n", "success");
	return 1;
}

# Alberta armor dealer (180,15) idxs: 0=Guard 3=Shoes 6=Muffler 16=Wooden Mail
# Buy only pieces still missing to avoid burning zeny / "Buy failed".
sub Grind_Phase2TalkBuyAlbertaArmor {
	my @buys;
	push @buys, 'b0,1' unless Grind_Phase2HasItem("Guard");
	push @buys, 'b3,1' unless Grind_Phase2HasItem("Shoes");
	push @buys, 'b6,1' unless Grind_Phase2HasRobe();
	push @buys, 'b16,1' unless Grind_Phase2HasItem("Wooden Mail");
	unless (@buys) {
		Log::message("[Grind] PHASE2 Alberta armor already owned — skip buy\n", "success");
		return 1;
	}
	my $seq = 'b ' . join(' ', @buys) . ' e';
	Commands::run("talknpc 180 15 $seq");
	Log::message("[Grind] PHASE2 Alberta buy: $seq\n", "success");
	return 1;
}

# payon_in01 7,119 armor shop indices (from npc_shops.txt):
# 1=Shoes(2403) 4=Muffler(2503) 13=Wooden Mail(2328)
# Prefer one-item talknpc from macros; this helper buys only still-missing pieces.
sub Grind_Phase2TalkBuyArmor {
	my $inv = $char && $char->inventory;
	my @seqs;
	push @seqs, 'b b1,1 e' unless $inv && $inv->getByName("Shoes");
	push @seqs, 'b b4,1 e' unless $inv && ($inv->getByName("Muffler") || $inv->getByName("Manteau"));
	push @seqs, 'b b13,1 e' unless $inv && $inv->getByName("Wooden Mail");
	unless (@seqs) {
		Log::message("[Grind] PHASE2 armor already owned\n", "success");
		return 1;
	}
	# Queue first missing piece only — macros call this repeatedly / buy one-by-one
	my $seq = $seqs[0];
	Commands::run("talknpc 7 119 $seq");
	Log::message("[Grind] PHASE2 talknpc buy armor: $seq\n", "success");
	return 1;
}

}
