# eventMacros.txt — Job Master + Blacksmith leveling route
# Novice → Merchant → Blacksmith  |  same map progression

# Job Master @ Prontera 150,180 (Job Master#ep2)
# First jobs: Swordman, Magician, Archer, Acolyte, Merchant, Thief
#   → Merchant = r4
# Merchant 2nd: Blacksmith, Alchemist
#   → Blacksmith = r0

automacro Ragna_Job_ToMerchant {
	JobID 0
	JobLevel >= 10
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Novice Job10+ → Prontera Job Master → Merchant
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c r4 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Merchant requested
	}
}

automacro Ragna_Job_ToBlacksmith {
	JobID 5
	JobLevel >= 40
	BaseLevel >= 40
	exclusive 1
	run-once 1
	timeout 60
	call {
		log [RagnaOne] Merchant Job40+ → Prontera Job Master → Blacksmith
		do conf attackAuto 0
		do conf route_randomWalk 0
		do move prontera 150 180
		pause 2
		do talknpc 150 180 c r0 c r0
		pause 3
		do conf attackAuto 2
		do conf route_randomWalk 1
		log [RagnaOne] Job change to Blacksmith requested
	}
}

automacro Ragna_AfterDeath {
	isAlive 0
	timeout 30
	exclusive 1
	call {
		[
			log [Ragna] Dead - no pause to avoid orphan macros
		]
	}
}

# ---- Town / Izlude recovery (RagnaOne) ----
automacro Ragna_TownAfterDeath {
	ConfigKeyNot lockMap none
	IsInMapAndCoordinate prontera,morocc,geffen,payon,alberta,izlude_in
	timeout 25
	exclusive 1
	call {
		[
			log [Ragna] In town after death - resume lockMap
			do reload conf
			pause 1
		]
	}
}

automacro Ragna_EscapeIzlude {
	IsInMapAndCoordinate izlude
	timeout 8
	exclusive 1
	call EscapeIzlude
}

macro EscapeIzlude {
	log [Ragna] Escaping Izlude step-walk west then south
	do west
	pause 2
	do west
	pause 2
	do west
	pause 2
	do west
	pause 2
	do west
	pause 2
	do south
	pause 2
	do south
	pause 2
	do south
	pause 2
	do west
	pause 2
	do west
	pause 2
}

automacro Ragna_EscapeIzludeIn {
	IsInMapAndCoordinate izlude_in
	timeout 8
	exclusive 1
	call {
		[
			log [Ragna] Leaving izlude_in toward outdoor Izlude
			do move 126 64
			pause 4
			do south
			pause 2
			do south
			pause 2
		]
	}
}
