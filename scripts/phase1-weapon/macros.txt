# =============================================================================
# Phase 1 — early leveling (CustomRO)
# =============================================================================
# Goal: farm Prontera fields until base ~30, auto-heal, sell/store, buy pots.
# At Job Level 10 (Novice): go to Prontera fountain, pick a random 1st class,
# equip job-change rewards, then resume grinding.
# Every 30 min: stop grind → random fountain spot → random facing → AFK 5–10 min.
#
# Start manually:   macro phase1
# Job change now:   macro phase1_changeJob
# Rent cart now:    macro phase1_rentCart
# AFK break now:    macro phase1_afkBreak
# Stop / reset:     macro phase1_stop
# Re-apply build:   macro phase1_applyBuild
#
# Auto stats/skills: raiseStat + raiseSkill plugins (statsAddAuto / skillsAddAuto).
# After job change the server resets stats → Phase1_ApplyBuild re-applies class build.
#   Mage: INT+DEX | Archer: DEX+AGI | Swordman: STR+AGI | etc.
# Merchant: learns Pushcart early, then Phase1_MerchantCart rents at fountain Rental Clerk.
#
# If the fountain NPC dialogue differs on your server, edit Phase1JobTalk
# and/or Phase1JobNPC_x / Phase1JobNPC_y below (set via phase1).
# =============================================================================

automacro Phase1_AutoStart {
	base <= 30
	timeout 30
	exclusive 1
	eval $::config{Phase} ne "Phase1" && $::config{Phase} ne "Phase1Done" && $::config{Phase} ne "Phase2" && $::config{Phase} ne "Phase2Done"
	call phase1
}

# Job Lv 10 Novice → Prontera fountain job change (random class)
automacro Phase1_ChangeJob {
	class Novice
	job >= 10
	timeout 20
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1JobDone} ne "1" && $::config{Phase1Busy} ne "1"
	call phase1_changeJob
}

# Keep free changejob weapons equipped (Blade / Composite Bow / Rod / Axe / Cutter / Chain…)
# Also re-runs after sellAuto used to strip rewards before equip fired.
automacro Phase1_EnsureJobWeapon {
	timeout 90
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{Phase1JobDone} eq "1" && $::char && $::char->inventory->isReady && !($::jobs_lut{$::char->{jobID}} eq "Novice")
	call phase1_ensureJobWeapon
}

# Apply/re-apply stats+skills when build flag does not match current class
# (covers Novice on start, and 1st class after changejob/stat reset/relog)
automacro Phase1_ApplyBuild {
	timeout 20
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::char && defined $::char->{jobID} && $::jobs_lut{$::char->{jobID}} && $::config{Phase1BuildApplied} ne $::jobs_lut{$::char->{jobID}}
	call phase1_applyBuild
}

# Safety: if Busy stuck past Phase1BusyUntil, force resume grind
automacro Phase1_UnstickBusy {
	timeout 60
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} eq "1" && $::config{Phase1BusyUntil} && time >= $::config{Phase1BusyUntil}
	call phase1_unstick
}

# Re-kick raiseStat/raiseSkill after they go INACTIVE (only if auto flags got turned off)
automacro Phase1_KickRaise {
	timeout 300
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && (($::config{statsAddAuto} ne "1" && $::config{statsAddAuto_list}) || ($::config{skillsAddAuto} ne "1" && $::config{skillsAddAuto_list}))
	call phase1_kickRaise
}

# After death / weird AI off: ensure combat is on while Phase1 farming
# Re-enable combat if something turned it off — but NEVER while town-healing
automacro Phase1_EnsureCombat {
	timeout 90
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{attackAuto} ne "2" && !($::field && $::field->isCity && $::char && $::char->{hp} < $::char->{hp_max} * 0.60)
	call phase1_ensureCombat
}

# Every ~30 minutes: AFK at fountain plaza only (mid Prontera, per-char corner)
# First AFK only after Phase1NextAfk timestamp (set when Phase1 starts).
# Never start AFK while dead/low-HP town-healing — that cancels lockMap and looks like "idle".
automacro Phase1_AfkBreak {
	timeout 60
	exclusive 1
	hp > 80%
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{Phase1NextAfk} && time >= $::config{Phase1NextAfk} && !$::char->{dead}
	call phase1_afkBreak
}

# If stuck in town too long while Phase1 farming, force return to lockMap
# Only leave when HP is safe — otherwise death-loops at save → field → die
automacro Phase1_LeaveTown {
	timeout 120
	exclusive 1
	hp > 60%
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{lockMap} && $::field && $::field->isCity && $::field->baseName ne $::config{lockMap}
	call phase1_leaveTown
}

# Low HP in town: sit and recover instead of walking out at 1–5% HP
automacro Phase1_TownHeal {
	timeout 30
	exclusive 1
	hp < 60%
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::field && $::field->isCity
	call phase1_townHeal
}

# After town sit-regen, restore combat flags so LeaveTown / field farming can run
automacro Phase1_TownHealDone {
	timeout 20
	exclusive 1
	hp > 80%
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::field && $::field->isCity && $::config{attackAuto} eq "0"
	call phase1_townHealDone
}

macro phase1_townHeal {
	[
		log Phase1 town heal — sit until HP safe (combat off while healing)
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf sitAuto_idle 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do sit
	]
}

macro phase1_townHealDone {
	[
		log Phase1 town heal done — combat back on
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf sitAuto_idle 1
		do stand
	]
}

macro phase1_leaveTown {
	[
		log Phase1 leave town — stand + scatter then lockMap
		# Keep buy enabled so low pots can route to Tool Dealer (prt_in 126,76)
		do conf buyAuto_0_npc prt_in 126 76
		do conf buyAuto_0_disabled 0
		do conf Phase1BuyPaused 0
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf sitAuto_idle 1
		do stand
	]
	# Per-char scatter near south gate so bots don't sit stacked
	$sx = @eval(140 + (ord(substr($::char->{name}, -1)) % 30))
	$sy = @eval(28 + (ord(substr($::char->{name}, 0, 1)) % 25))
	log Phase1 leave scatter $sx $sy (char-based)
	do move $sx $sy
	pause 2
	call phase1_setLock
	[
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf sitAuto_idle 1
	]
	# Explicitly route to lockMap — otherwise LeaveTown re-scatters every 120s
	# while still in town and never exits Prontera.
	do move @config(lockMap)
	do ai on
}

# If stuck outside lockMap (and not in town), re-apply lock so AI walks back
automacro Phase1_ReturnToField {
	base <= 30
	timeout 45
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{lockMap} && $::field && !$::field->isCity && $::field->baseName ne $::config{lockMap}
	call phase1_setLock
}

# Advance lock map within Phase 1 as you level
automacro Phase1_LevelBand {
	base > 0
	timeout 60
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1"
	call phase1_setLock
}

# Low Red Potions + have zeny → ensure buyAuto points at Tool Dealer
automacro Phase1_LowPots {
	inventory "Red Potion" < 5
	inventory "Novice Potion" < 5
	zeny >= 50
	base <= 30
	timeout 300
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1"
	call phase1_buyPots
}

# No zeny → disable all buyAuto (ignore shopping until we can afford it)
automacro Phase1_NoZenySkipBuy {
	zeny < 50
	timeout 60
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{buyAuto_0_disabled} ne "1"
	call phase1_disableBuy
}

# Got enough zeny again → re-enable Tool Dealer buy
automacro Phase1_HasZenyEnableBuy {
	zeny >= 50
	timeout 300
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{Phase1BuyPaused} eq "1"
	call phase1_enableBuy
}

# Merchant without cart → Prontera fountain Rental Clerk (needs Pushcart skill)
automacro Phase1_MerchantCart {
	class Merchant
	timeout 90
	exclusive 1
	eval $::config{Phase} eq "Phase1" && $::config{Phase1Busy} ne "1" && $::config{Phase1CartDone} ne "1" && $::char && !$::char->cartActive && (($::char->{skills}{MC_PUSHCART}{lv} || 0) >= 1)
	call phase1_rentCart
}

macro phase1_buyPotsSetup {
	[
		# RagnaOne Tool Dealer is INSIDE the north Prontera shop (not outdoor 57/64,125).
		# Manual shop: portal prontera 134,221 → prt_in → Tool Dealer#prt1 (126,76).
		# Red Potion 50z confirmed 2026-08-16.
		do conf buyAuto_0 Red Potion
		do conf buyAuto_0_npc prt_in 126 76
		do conf buyAuto_0_npc_steps b
		do conf buyAuto_0_minAmount 10
		do conf buyAuto_0_maxAmount 30
		do conf buyAuto_0_price 50
		do conf buyAuto_0_zeny >= 50
		do conf buyAuto_0_disabled 0
		do conf Phase1BuyPaused 0
		do conf minStorageZeny 50
		# Sell at Butcher near west market
		do conf sellAuto_npc prontera 64 125
	]
}

macro phase1_disableBuy {
	[
		log Phase1 — not enough zeny ($.zeny), skip buying pots/items
		do conf buyAuto_0_disabled 1
		do conf Phase1BuyPaused 1
	]
}

macro phase1_enableBuy {
	[
		log Phase1 — zeny ok ($.zeny), buyAuto ON (Tool Dealer prt_in 126,76)
		do conf buyAuto_0 Red Potion
		do conf buyAuto_0_npc prt_in 126 76
		do conf buyAuto_0_npc_steps b
		do conf buyAuto_0_disabled 0
		do conf Phase1BuyPaused 0
	]
}

macro phase1_setLock {
	$blvl = $.lvl
	# Stay on prt_fild08 until Phase2 (base 30). Knife chars deal ~3 to Poporing on fild07.
	if ($blvl < 30) {
		do conf lockMap prt_fild08
		do conf lockMap_x none
		do conf lockMap_y none
		do mconf Poporing 0 0 0
		do mconf Rocker 0 0 0
		do mconf Poring 2 0 0
		do mconf Lunatic 2 0 0
		do mconf Drops 2 0 0
		do mconf 1007 2 0 0
		do mconf Pupa 2 0 0
		log Phase1 lock prt_fild08 (base $blvl) — avoid Poporing until Phase2
	} else {
		do conf lockMap pay_fild08
		do conf lockMap_x none
		do conf lockMap_y none
		do mconf Poporing 2 0 0
		do mconf Rocker 1 0 0
		do mconf Spore 2 0 0
		do mconf Willow 2 0 0
		log Phase1 lock pay_fild08 (base $blvl) — handoff to Phase2
	}
}

# -----------------------------------------------------------------------------
# Merchant cart rental at Prontera fountain (Rental Clerk)
# -----------------------------------------------------------------------------
macro phase1_rentCart {
	[
		log ========================================
		log Phase1 CART RENTAL — Merchant has Pushcart, no cart
		log Going to Prontera Rental Clerk (fountain)
		log ========================================
		do conf Phase1Busy 1
		do conf Phase1BusyUntil @eval(time + 600)
		do ai manual
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
	]
	pause 1

	$cx = @config(Phase1CartNPC_x)
	$cy = @config(Phase1CartNPC_y)
	$ctalk = @config(Phase1CartTalk)
	do move prontera $cx $cy
	$wait = 0

	:cartWaitPron
	if ($.map == prontera) goto cartAtPron
	$wait++
	if ($wait > 90) goto cartFail
	pause 5
	goto cartWaitPron

	:cartAtPron
	pause 2
	do move prontera $cx $cy
	pause 3

	log Phase1 talk Rental Clerk at $cx $cy steps: $ctalk
	do talknpc $cx $cy $ctalk
	pause 3

	# Fallback: continue-only (some servers auto-rent cart with no menu)
	do talknpc $cx $cy c
	pause 2

	:cartCheck
	[
		do conf storageAuto 1
		do conf sellAuto 1
		do conf buyAuto_0_disabled 0
		do conf Phase1BuyPaused 0
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf sitAuto_idle 1
		do conf Phase1Busy 0
		do conf Phase1BusyUntil
	]
	# Mark done only if cart is actually active — else retry later
	$hascart = @eval($::char && $::char->cartActive ? 1 : 0)
	if ($hascart == 1) goto cartOk
	[
		log Phase1 cart rental FAILED — will retry when Pushcart/NPC ok
		do conf Phase1CartDone 0
	]
	goto cartResume

	:cartOk
	[
		log Phase1 cart rental OK
		do conf Phase1CartDone 1
	]
	goto cartResume

	:cartFail
	[
		log Phase1 cart rental aborted — never reached Prontera
		do conf Phase1Busy 0
		do conf Phase1BusyUntil
		do conf Phase1CartDone 0
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf storageAuto 1
		do conf sellAuto 1
	]

	:cartResume
	call phase1_setLock
	do ai on
	log Phase1 cart rental END — back to farming
}

# -----------------------------------------------------------------------------
# Job change at Prontera fountain (random 1st class) + equip rewards
# -----------------------------------------------------------------------------
macro phase1_changeJob {
	[
		log ========================================
		log Phase1 JOB CHANGE — Job Lv $.joblvl Novice
		log Going to Prontera fountain (random class)
		log ========================================
		do conf Phase1Busy 1
		do conf Phase1BusyUntil @eval(time + 1200)
		do ai manual
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf storageAuto 0
		do conf sellAuto 0
		do conf buyAuto_0_disabled 1
	]
	pause 1

	# Walk to fountain / Job Master and WAIT until we are actually in Prontera
	$jx = @config(Phase1JobNPC_x)
	$jy = @config(Phase1JobNPC_y)
	do move prontera $jx $jy
	$wait = 0

	:waitPron
	if ($.map == prontera) goto atPron
	$wait++
	if ($wait > 90) goto jobFail
	pause 5
	goto waitPron

	:atPron
	pause 2
	do move prontera $jx $jy
	pause 3

	# Random first class (explicit — @listitem is unreliable in this OpenKore build)
	$pick = @rand(0, 5)
	if ($pick == 0) goto setSword
	if ($pick == 1) goto setMage
	if ($pick == 2) goto setArcher
	if ($pick == 3) goto setAco
	if ($pick == 4) goto setMerch
	goto setThief

	:setSword
	$jobname = Swordman
	goto setDone
	:setMage
	$jobname = Mage
	goto setDone
	:setArcher
	$jobname = Archer
	goto setDone
	:setAco
	$jobname = Acolyte
	goto setDone
	:setMerch
	$jobname = Merchant
	goto setDone
	:setThief
	$jobname = Thief
	goto setDone

	:setDone
	do conf Phase1PickedJob $jobname
	log Phase1 random class pick: $jobname (menu index $pick)

	# RagnaOne Job Master: select class → confirm (no intermediate ChangeJob menu)
	do talknpc $jx $jy r$pick r0
	pause 3
	# Still Novice at Job10+ → retry with name match + confirm
	if ($.joblvl >= 10) goto tryNamed
	goto afterTalk

	:tryNamed
	log Phase1 talk retry with class name: $jobname
	do talknpc $jx $jy r~/Swordsman|Magician|Archer|Acolyte|Merchant|Thief/i r0
	pause 3

	:afterTalk
	# Abort cleanly if still Novice (do not mark JobDone)
	if ($.joblvl >= 10) goto jobFail
	# Rewards often arrive a few seconds after the job packet — wait, then equip
	# BEFORE re-enabling sellAuto (Composite Bow / Blade were being sold).
	[
		do conf sellAuto 0
		do conf storageAuto 0
		log Phase1 after talk — waiting for job-reward items then equipping
	]
	pause 5
	call phase1_equipRewards
	pause 2
	call phase1_equipRewards
	pause 1

	# Server resets stats after changejob — clear build flag, wait, re-apply class build
	[
		do conf Phase1JobDone 1
		do conf Phase1WeaponDone 1
		do conf Phase1BuildApplied
		do conf statsAddAuto 0
		do conf skillsAddAuto 0
	]
	pause 3
	call phase1_applyBuild
	pause 1
	goto jobOk

	:jobFail
	[
		log Phase1 JOB CHANGE FAILED — still Novice or never reached Prontera; retry later
		do talk
		do conf Phase1JobDone 0
		do conf Phase1Busy 0
		do conf Phase1BusyUntil
		do conf storageAuto 1
		do conf sellAuto 1
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
	]

# Periodic: equip free job weapons still sitting unequipped; buy replacement if already sold.
macro phase1_ensureJobWeapon {

	if (@config(Phase1WeaponBuyNow) == 1) goto ewBuy
	goto ewStop

	goto ewWaitMap

	do move prt_church 108 122

	:ewWaitMap

	:ewWaitCont
	$w++
	if ($w > 50) goto ewBuyDone
	pause 5
	goto ewWaitLoop

		do conf route_randomWalk_inLockOnly 1

# -----------------------------------------------------------------------------
# AFK break: every 30 min → Prontera fountain (mid town), per-char random tile + facing
# -----------------------------------------------------------------------------
macro phase1_afkBreak {
	[
		log ========================================
		log Phase1 AFK BREAK — fountain plaza (mid Prontera)
		log ========================================
		do conf Phase1Busy 1
		do conf Phase1BusyUntil @eval(time + 1200)
		do ai manual
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf storageAuto 0
		do conf sellAuto 0
		do conf buyAuto_0_disabled 1
		do conf sitAuto_idle 0
		do stand
	]
	pause 1

	# Fountain / middle Prontera only (avoid fountain water ~156,198).
	# 4 corners around the plaza from char name so bots don't stack on one tile.
	$zone = @eval( (ord(substr($::char->{name}, 0, 1)) + ord(substr($::char->{name}, -1)) + length($::char->{name})) % 4 )
	if ($zone == 0) goto afkZ0
	if ($zone == 1) goto afkZ1
	if ($zone == 2) goto afkZ2
	goto afkZ3

	:afkZ0
	# SW of fountain
	$ax = @rand(148, 154)
	$ay = @rand(172, 182)
	goto afkGo

	:afkZ1
	# SE of fountain
	$ax = @rand(158, 164)
	$ay = @rand(172, 182)
	goto afkGo

	:afkZ2
	# NW of fountain
	$ax = @rand(148, 154)
	$ay = @rand(186, 194)
	goto afkGo

	:afkZ3
	# NE of fountain
	$ax = @rand(158, 164)
	$ay = @rand(186, 194)
	goto afkGo

	:afkGo
	# Small per-char jitter within the corner
	$ox = @eval( (ord(substr($::char->{name}, -1)) % 5) - 2 )
	$oy = @eval( (ord(substr($::char->{name}, 0, 1)) % 5) - 2 )
	$ax = @eval( $ax + $ox )
	$ay = @eval( $ay + $oy )
	log Phase1 AFK fountain corner $zone walk to prontera $ax $ay
	do move prontera $ax $ay
	$wait = 0

	:afkWaitMap
	if ($.map == prontera) goto afkAtPron
	$wait++
	if ($wait > 90) goto afkAbort
	pause 5
	goto afkWaitMap

	:afkAtPron
	do move $ax $ay
	pause 8
	do move $ax $ay
	pause 4

	$body = @rand(0, 7)
	$head = @rand(0, 2)
	do look $body $head
	log Phase1 AFK facing body=$body head=$head at $.posx $.posy
	pause 1
	do sit

	$afk = @rand(300, 600)
	log Phase1 AFK idle for $afk seconds
	pause $afk

	$body2 = @rand(0, 7)
	$head2 = @rand(0, 2)
	do look $body2 $head2
	pause 2
	do stand
	pause 1
	goto afkDone

	:afkAbort
	[
		log Phase1 AFK aborted — never reached Prontera
		do conf Phase1Busy 0
		do conf Phase1BusyUntil
		do conf Phase1NextAfk @eval(time + 1800 + (ord(substr($::char->{name}, -1)) % 8) * 90)
		do conf storageAuto 1
		do conf sellAuto 1
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_idle 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
	]
	call phase1_setLock
	do ai on
	stop

	:afkDone
	[
		log Phase1 AFK done — resuming grind
		do conf Phase1Busy 0
		do conf Phase1BusyUntil
		do conf storageAuto 1
		do conf sellAuto 1
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_idle 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf Phase1NextAfk @eval(time + 1800 + (ord(substr($::char->{name}, -1)) % 8) * 90)
	]
	if ($.zeny < 50) goto afkNobuy
	[
		do conf buyAuto_0_disabled 0
		do conf Phase1BuyPaused 0
	]
	goto afkResume

	:afkNobuy
	[
		do conf buyAuto_0_disabled 1
		do conf Phase1BuyPaused 1
		log Phase1 AFK done — no zeny, skip buying
	]

	:afkResume
	call phase1_setLock
	do ai on
	log ========================================
	log Phase1 AFK BREAK END — back to farming
	log ========================================
}

	]

}

macro phase2_setLock {
	$blvl = $.lvl
	if ($blvl < 40) {
		do conf lockMap pay_fild08
		do conf lockMap_x none
		do conf lockMap_y none
		log Phase2 lock pay_fild08 (base $blvl)
	} else {
		if ($blvl < 50) {
			do conf lockMap pay_fild09
			do conf lockMap_x none
			do conf lockMap_y none
			log Phase2 lock pay_fild09 (base $blvl)

			do conf lockMap moc_fild02
			do conf lockMap_x none
			do conf lockMap_y none
			log Phase2 lock moc_fild02 (base $blvl)
		}
	}
	call phase2_applyMonsters
}

macro phase2_rentCart {
	[
		log Phase2 CART RENTAL — Merchant Pushcart, no cart
		do conf Phase2Busy 1
		do conf Phase2BusyUntil @eval(time + 600)
		do ai manual
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
	]
	pause 1
	$cx = @config(Phase1CartNPC_x)
	$cy = @config(Phase1CartNPC_y)
	$ctalk = @config(Phase1CartTalk)
	do move prontera $cx $cy
	$wait = 0
	:p2cartWait
	if ($.map == prontera) goto p2cartAt
	$wait++
	if ($wait > 90) goto p2cartFail
	pause 5
	goto p2cartWait
	:p2cartAt
	pause 2
	do move prontera $cx $cy
	pause 3
	do talknpc $cx $cy $ctalk
	pause 3
	do talknpc $cx $cy c
	pause 2
	[
		do conf storageAuto 0
		do conf sellAuto 1
		do conf buyAuto_0_disabled 0
		do conf Phase2BuyPaused 0
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf Phase2Busy 0
		do conf Phase2BusyUntil
	]
	$hascart = @eval($::char && $::char->cartActive ? 1 : 0)
	if ($hascart == 1) goto p2cartOk
	[
		log Phase2 cart rental FAILED
		do conf Phase2CartDone 0
	]
	goto p2cartResume
	:p2cartOk
	[
		log Phase2 cart rental OK
		do conf Phase2CartDone 1
	]
	goto p2cartResume
	:p2cartFail
	[
		log Phase2 cart rental aborted
		do conf Phase2Busy 0
		do conf Phase2BusyUntil
		do conf Phase2CartDone 0
		do conf attackAuto 2
		do conf route_randomWalk 1
	]
	:p2cartResume
	call phase2_setLock
	do ai on
}

macro phase2_afkBreak {
	[
		log Phase2 AFK BREAK — fountain plaza
		do conf Phase2Busy 1
		do conf Phase2BusyUntil @eval(time + 1200)
		do ai manual
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf storageAuto 0
		do conf sellAuto 0
		do conf buyAuto_0_disabled 1
		do conf sitAuto_idle 0
		do stand
	]
	pause 1
	$zone = @eval( (ord(substr($::char->{name}, 0, 1)) + ord(substr($::char->{name}, -1)) + length($::char->{name})) % 4 )
	if ($zone == 0) goto p2afkZ0
	if ($zone == 1) goto p2afkZ1
	if ($zone == 2) goto p2afkZ2
	goto p2afkZ3
	:p2afkZ0
	$ax = @rand(148, 154)
	$ay = @rand(172, 182)
	goto p2afkGo
	:p2afkZ1
	$ax = @rand(158, 164)
	$ay = @rand(172, 182)
	goto p2afkGo
	:p2afkZ2
	$ax = @rand(148, 154)
	$ay = @rand(186, 194)
	goto p2afkGo
	:p2afkZ3
	$ax = @rand(158, 164)
	$ay = @rand(186, 194)
	:p2afkGo
	$ox = @eval( (ord(substr($::char->{name}, -1)) % 5) - 2 )
	$oy = @eval( (ord(substr($::char->{name}, 0, 1)) % 5) - 2 )
	$ax = @eval( $ax + $ox )
	$ay = @eval( $ay + $oy )
	log Phase2 AFK walk prontera $ax $ay
	do move prontera $ax $ay
	$wait = 0
	:p2afkWait
	if ($.map == prontera) goto p2afkAt
	$wait++
	if ($wait > 90) goto p2afkAbort
	pause 5
	goto p2afkWait
	:p2afkAt
	do move $ax $ay
	pause 8
	$body = @rand(0, 7)
	$head = @rand(0, 2)
	do look $body $head
	do sit
	$afk = @rand(300, 600)
	log Phase2 AFK idle $afk seconds
	pause $afk
	do stand
	pause 1
	goto p2afkDone
	:p2afkAbort
	[
		log Phase2 AFK aborted
		do conf Phase2Busy 0
		do conf Phase2BusyUntil
		do conf Phase2NextAfk @eval(time + 1800)
		do conf storageAuto 0
		do conf sellAuto 1
		do conf attackAuto 2
		do conf route_randomWalk 1
	]
	call phase2_setLock
	do ai on
	stop
	:p2afkDone
	[
		log Phase2 AFK done
		do conf Phase2Busy 0
		do conf Phase2BusyUntil
		do conf storageAuto 0
		do conf sellAuto 1
		do conf attackAuto 2
		do conf route_randomWalk 1
		do conf route_randomWalk_inLockOnly 1
		do conf sitAuto_idle 1
		do conf sitAuto_hp_lower 20
		do conf sitAuto_hp_upper 90
		do conf Phase2NextAfk @eval(time + 1800 + (ord(substr($::char->{name}, -1)) % 8) * 90)
	]
	if ($.zeny < 50) goto p2afkNobuy
	[
		do conf buyAuto_0_disabled 0
		do conf Phase2BuyPaused 0
	]
	goto p2afkResume
	:p2afkNobuy
	[
		do conf buyAuto_0_disabled 1
		do conf Phase2BuyPaused 1
	]
	:p2afkResume
	call phase2_setLock
	do ai on
	log Phase2 AFK END — farming
}
