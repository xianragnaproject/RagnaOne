macro phase2_payon_afk {
	[
		log PHASE2: AFK break - Payon middle Kafra then sit
		do conf phase2Afk 1
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sitAuto_idle 0

		do eval AI::clear("move", "route", "mapRoute", "attack")
	]
	call phase2_to_payon
	[
		do eval Phase2_WalkKafraAFK()
	]
	pause 55
	call phase2_afk_face_sit
	$secs = &rand(300, 900)
	log PHASE2: AFK sit-idle for $secs sec
	pause $secs
	[
		do stand
		log PHASE2: AFK done - resume hunt
		do conf sitAuto_idle 1
		do conf phase2Afk 0
		do eval Phase2_ResumeHunt()
	]
}

# Get to a city Kafra → Teleport Service → Payon.

macro phase2_to_payon {
	[
		log PHASE2: go Payon via Kafra teleport service (not field walk)
		do conf lockMap none
		do conf route_randomWalk 0
		do conf attackAuto 0
		do conf sitAuto_idle 0
		do stand
		do eval Phase2_DumpJunkWeight()
		do eval Phase2_GoPayon()
	]
	if (&config(phase2InPayon) == 1) goto phase2_town_ready
	pause 40
	[
		do eval Phase2_GoPayon()
	]
	pause 45
	[
		do eval Phase2_GoPayon()
	]
	pause 45
	[
		do eval Phase2_GoPayon()
	]
	pause 40
	[
		do eval Phase2_GoPayon()
	]
	pause 30
	if (&config(phase2InPayon) == 1) goto phase2_town_ready
	[
		log PHASE2: still not in Payon - will retry next automacro
		do conf phase2Traveling 0
	]
	stop
	:phase2_town_ready
	[
		log PHASE2: Payon ready
		do conf phase2Traveling 0
	]
}

# ==== perl helpers (eventMacro appends to shared interpreter) ====
sub Phase2_IsInPayonTown {
	return 0 unless $field;
	my $m = $field->baseName // '';
	return ($m eq 'payon') ? 1 : 0;
}

# Drop common junk so OW drops below move penalties before town travel
sub Phase2_DumpJunkWeight {
	return 0 unless $char && $char->inventory;
	my $pct = 0;
	eval { $pct = Misc::percent_weight($char) };
	return 1 if $pct < 50;
	my @junk = (
		'Jellopy', 'Empty Bottle', 'Feather', 'Fluff', 'Apple', 'Carrot',
		'Clover', 'Shell', 'Sticky Mucus', 'Chrysalis', 'Stone', 'Unripe Apple',
	);
	my $dropped = 0;
	foreach my $name (@junk) {
		last if Misc::percent_weight($char) < 45;
		my $item = $char->inventory->getByName($name);
		next unless $item && ($item->{amount} // 0) > 0;
		my $n = $item->{amount};
		Commands::run("drop $item->{binID} $n");
		$dropped += $n;
		select(undef, undef, undef, 0.35);
	}
	Log::message("[PHASE2] dumped junk (approx $dropped) weight now ".Misc::percent_weight($char)."%\n", "success");
	return 1;
}

# Prontera Kafra → Payon Teleport Service (openkore portals.txt: c r2 c r2)
# Kafras: south 151,29 | mid 146,89 | east 282,200 | west 35,208
sub Phase2_TalkKafraWarpPayon {
	my @kafras = (
		[151, 29],
		[146, 89],
		[282, 200],
		[35, 208],
	);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;
	my ($best, $bd);
	foreach my $k (@kafras) {
		my $d = abs($x - $k->[0]) + abs($y - $k->[1]);
		if (!defined $bd || $d < $bd) {
			$bd = $d;
			$best = $k;
		}
	}
	return 0 unless $best;
	# Only talk if reasonably close (else walk first)
	if ($bd > 16) {
		Log::message("[PHASE2] walk to Kafra $best->[0],$best->[1] for Payon teleport\n", "success");
		Commands::run("move $best->[0] $best->[1]");
		return 0;
	}
	Log::message("[PHASE2] Teleport Service → Payon via Kafra $best->[0],$best->[1]\n", "success");
	Commands::run("talknpc $best->[0] $best->[1] c r2 c r2");
	return 1;
}

sub Phase2_GoPayon {
	Commands::run("stand") if $char && $char->{sitting};
	my $m = $field ? ($field->baseName // '') : '';
	if ($m eq 'payon') {
		Commands::run("conf -f phase2InPayon 1");
		Log::message("[PHASE2] already in Payon\n", "success");
		return 1;
	}
	Commands::run("conf -f phase2InPayon 0");

	# Do not interrupt an in-progress walk/route (was causing permanent Stuck)
	if (AI::inQueue("route") || AI::inQueue("mapRoute") || AI::inQueue("move")
		|| AI::is("route", "mapRoute", "move") || AI::is("NPC")) {
		Log::message("[PHASE2] already routing/NPC — let it finish\n", "system");
		return 0;
	}

	# From hunt map: only the short town portal (not field-hopping to Prontera)
	if ($m eq 'pay_fild08') {
		Log::message("[PHASE2] pay_fild08 → Payon town portal 17,75\n", "success");
		Commands::run("move 17 75");
		return 0;
	}

	# From Prontera: walk inside city to Kafra, then Teleport Service → Payon
	# NEVER "move payon ..." (that walks through fields)
	if ($m eq 'prontera') {
		return Phase2_TalkKafraWarpPayon();
	}

	# From Prontera field: enter Prontera city, then Kafra teleport (not walk to Payon)
	if ($m eq 'prt_fild08') {
		Log::message("[PHASE2] prt_fild08 → Prontera city then Kafra teleport to Payon\n", "success");
		Commands::run("move 170 378");
		return 0;
	}

	# Elsewhere: go Prontera south Kafra only — next ticks use Teleport Service
	Log::message("[PHASE2] go Prontera Kafra then Teleport Service to Payon\n", "success");
	Commands::run("move prontera 151 29");
	return 0;
}

sub Phase2_WalkKafraAFK {
	unless (Phase2_IsInPayonTown()) {
		Log::message("[PHASE2] not in Payon — use Kafra teleport (no field walk)\n", "system");
		Phase2_GoPayon();
		return 0;
	}
	my $x = 175 + int(rand(13));   # 175-187 near middle kafra
	my $y = 98 + int(rand(13));    # 98-110
	Log::message("[PHASE2] kafra AFK walk → $x,$y\n", "success");
	Commands::run("stand") if $char && $char->{sitting};
	Commands::run("move $x $y");
	return 1;
}

sub Phase2_StepFaceKafra {
	return 0 unless $char && Phase2_IsInPayonTown();
	Commands::run("stand") if $char->{sitting};
	AI::clear("move", "route", "mapRoute", "sitting", "standing", "look");
	my @dx = (0, -1, -1, -1, 0, 1, 1, 1);
	my @dy = (1, 1, 0, -1, -1, -1, 0, 1);
	my @dir = qw(N NW W SW S SE E NE);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 181;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 104;
	my @cands;
	for my $d (0 .. 7) {
		my ($tx, $ty) = ($x + $dx[$d], $y + $dy[$d]);
		next unless abs($tx - 181) <= 18 && abs($ty - 104) <= 18;
		push @cands, [$d, $tx, $ty];
	}
	unless (@cands) {
		for my $d (0 .. 7) {
			push @cands, [$d, $x + $dx[$d], $y + $dy[$d]];
		}
	}
	for (my $i = @cands - 1; $i > 0; $i--) {
		my $j = int(rand($i + 1));
		@cands[$i, $j] = @cands[$j, $i];
	}
	my ($body, $tx, $ty) = @{$cands[0]};
	Commands::run("conf sitAuto_look $body");
	Commands::run("conf sitAuto_look_delay 0");
	Commands::run("move $tx $ty");
	Log::message("[PHASE2] AFK step face=$body ($dir[$body]) → $tx,$ty\n", "success");
	return $body;
}

sub Phase2_AfkSit {
	return 1 if $char && $char->{sitting};
	unless (Phase2_IsInPayonTown()) {
		Log::message("[PHASE2] AFK sit skipped - not in Payon\n", "system");
		return 0;
	}
	my $basic = 0;
	if ($char && $char->{skills}{NV_BASIC}) {
		$basic = $char->{skills}{NV_BASIC}{lv} // 0;
	}
	if ($basic < 3) {
		Log::message("[PHASE2] AFK stand-idle - Basic Skill $basic < 3\n", "system");
		return 0;
	}
	$ai_v{sitAuto_forcedBySitCommand} = 1;
	$ai_v{sitAuto_forceStop} = 0;
	Commands::run("sit");
	return 1;
}

sub Phase2_ApplyJobBuild {
	my %id_to_job = (
		1 => 'swordman', 2 => 'magician', 3 => 'archer',
		4 => 'acolyte', 5 => 'merchant', 6 => 'thief',
	);
	my $job = lc($config{grindTargetJob} || '');
	if ($job eq '' || $job eq 'random' || $job eq 'none' || $job eq 'auto') {
		$job = $id_to_job{ $char->{jobID} // -1 } // 'swordman';
		Commands::run("conf grindTargetJob $job");
	}
	$job = 'swordman' if $job eq 'swordsman';
	$job = 'magician' if $job eq 'mage';

	# Pure two-stat builds through mid levels
	my %stats = (
		swordman => '20 str, 20 agi, 40 str, 40 agi, 60 str, 60 agi, 80 str, 80 agi, 90 str, 90 agi',
		magician => '20 int, 20 dex, 40 int, 40 dex, 60 int, 60 dex, 80 int, 80 dex, 90 int, 90 dex',
		archer   => '20 dex, 20 agi, 40 dex, 40 agi, 60 dex, 60 agi, 80 dex, 80 agi, 90 dex, 90 agi',
		acolyte  => '20 int, 20 dex, 40 int, 40 dex, 60 int, 60 dex, 80 int, 80 dex, 90 int, 90 dex',
		merchant => '20 str, 20 dex, 40 str, 40 dex, 60 str, 60 dex, 80 str, 80 dex, 90 str, 90 dex',
		thief    => '20 agi, 20 dex, 40 agi, 40 dex, 60 agi, 60 dex, 80 agi, 80 dex, 90 agi, 90 dex',
	);
	my %skills = (
		swordman => 'Basic Skill 9, Bash 10, Provoke 5, Endure 5, Magnum Break 10',
		magician => 'Basic Skill 9, Cold Bolt 10, Fire Bolt 10, Lightning Bolt 10, Increase SP Recovery 10',
		archer   => 'Basic Skill 9, Owl\'s Eye 10, Vulture\'s Eye 10, Double Strafe 10, Arrow Shower 5',
		acolyte  => 'Basic Skill 9, Divine Protection 10, Demon Bane 5, Heal 10, Increase AGI 10, Blessing 10',
		merchant => 'Basic Skill 9, Enlarge Weight Limit 10, Discount 10, Overcharge 10, Pushcart 10, Vending 5',
		thief    => 'Basic Skill 9, Double Attack 10, Improve Dodge 10, Stealing 10, Envenom 10, Hiding 5',
	);

	my $sk = $skills{$job} // $skills{swordman};
	my $st = $stats{$job} // $stats{swordman};
	Commands::run("conf -f skillsAddAuto_list $sk");
	Commands::run("conf -f skillsAddAuto 1");
	Commands::run("conf -f statsAddAuto_list $st");
	Commands::run("conf -f statsAddAuto 1");
	Commands::run("conf -f statsAddAuto_dontUseBonus 0");
	Log::message("[PHASE2] Auto skill/stat build for $job (pure)\n", "success");
	return 1;
}
