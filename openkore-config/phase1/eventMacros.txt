###############################################################################
# RagnaOne Phase 1 macros (Novice base 1-15)
#
# Macros own config — they set lockMap / sellAuto / buyAuto / saveMap themselves.
#
# 1) LockMap prt_fild08          — Novice base 1-15
# 2) Save Prontera south Kafra   — Novice base 1-15
# 3) Sell @ Prontera tool dealer — Novice base 1-15, OW >= 50%
# 4) Buy 50 Red Potion           — Novice, zeny >= 5000, stock 0
# 5) Fountain AFK break          — ~every 30 min, sit 5-15 min, random facing
# 6) Job change @ Job Lv 10      — random 1st job at Prontera Job Master
# 7) Base 15 + jobbed            — park/sit near Prontera fountain (random face)
#
# Flags (set by macros):
#   phase1LockDone  phase1SaveDone  phase1ShopDone  phase1Jobbing
#   phase1JobDone  phase1Afk  phase1Done  grindTargetJob
###############################################################################


###############################################################################
# 1) LockMap -> prt_fild08 (Novice base 1-15) — RUN ONCE
###############################################################################
automacro Phase1_SetLockMap {
	exclusive 1
	run-once 1
	timeout 30
	priority 4
	JobID 0
	BaseLevel 1..15

	ConfigKeyNot phase1Done 1
	ConfigKeyNot phase1Afk 1
	ConfigKeyNot phase1Jobbing 1
	ConfigKeyNot phase1Selling 1
	Eval !($char->{lv_job} >= 10)
	call {
		[
			log PHASE1: set lockMap prt_fild08 Novice 1-15 once
			do conf lockMap prt_fild08
			do conf lockMap_x none
			do conf lockMap_y none

			do conf attackMinPlayerDistance 0
			do conf route_randomWalk 1
			do conf route_randomWalk_inLockOnly 1
			do conf -f phase1LockDone 1
		]
	}
}

###############################################################################
# 5) Fountain AFK — ~every 30 min, sit 5-15 min, random facing (not always N)
###############################################################################
automacro Phase1_FountainAFK {
	exclusive 1
	timeout 1800
	priority 6
	InMap prt_fild08
	JobID 0
	BaseLevel 1..15
	ConfigKeyNot phase1Done 1
	ConfigKeyNot phase1Afk 1
	ConfigKeyNot phase1Selling 1
	ConfigKeyNot phase1Jobbing 1
	Eval !($char && $char->{lv_job} >= 10)
	call phase1_fountain_afk
}

macro phase1_fountain_afk {
	$secs = &rand(300, 900)
	[
		log PHASE1: AFK break - fountain for $secs sec 5-15 min
		do conf phase1Afk 1
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sitAuto_idle 0
		do stand
		do eval AI::clear("move", "route", "mapRoute", "attack")
	]
	call phase1_to_prontera
	[
		do eval Phase1_WalkFountain()
		pause 50
	]
	call phase1_afk_face_sit
	[
		pause $secs
		do stand
		log PHASE1: AFK done — resume hunt
		do conf sitAuto_idle 1
		do conf phase1Afk 0
		do eval Phase1_ResumeHunt()
	]
}

# One random step (sets visible body facing), then sit
macro phase1_afk_face_sit {
	[
		do eval Phase1_StepFace()
		pause 8
		do eval Phase1_AfkSit()
		pause 4
	]
}

		do conf lockMap none

###############################################################################
# Perl helpers
###############################################################################
sub Phase1_IsInPronteraTown {
	return 0 unless $field;
	my $m = $field->baseName;
	return ($m eq 'prontera' || $m eq 'prt_in') ? 1 : 0;
}

sub Phase1_ApproachFieldPortal {
	return 1 if Phase1_IsInPronteraTown();
	return 1 unless $field && $field->baseName eq 'prt_fild08';
	my $x = $char->{pos}{x} // 0;
	my $y = $char->{pos}{y} // 0;
	return 1 if abs($x - 170) <= 2 && abs($y - 375) <= 3;
	Commands::run("move 170 375");
	return 1;
}

sub Phase1_EnterFieldPortal {
	return 1 if Phase1_IsInPronteraTown();
	return 1 unless $field && $field->baseName eq 'prt_fild08';
	my $x = $char->{pos}{x} // 0;
	my $y = $char->{pos}{y} // 0;
	unless (abs($x - 170) <= 6 && abs($y - 375) <= 10) {
		Commands::run("stand") if $char->{sitting};
		Commands::run("move 170 375");
		return 1;
	}
	Commands::run("stand") if $char->{sitting};
	if ($y < 377) {
		Commands::run("move 170 377");
	} elsif ($y < 378) {
		Commands::run("north");
	} else {
		Commands::run("move prontera 156 30");
	}
	return 1;
}

sub Phase1_RandomFountainTile {
	my @spots = (
		[155, 185], [150, 185], [160, 185], [148, 185], [162, 185],
		[155, 180], [152, 182], [158, 182], [150, 182], [160, 182],
		[155, 188], [152, 188], [158, 188], [148, 188], [162, 188],
		[155, 190], [150, 190], [160, 190], [152, 190], [158, 190],
		[145, 192], [146, 188], [166, 192], [167, 188],
		[145, 185], [167, 185],
	);
	my @ok;
	if ($field && $field->baseName eq 'prontera') {
		for my $s (@spots) {
			next unless $field->isWalkable($s->[0], $s->[1]);
			push @ok, $s;
		}
	}
	@ok = (
		[155, 185], [150, 185], [160, 185], [152, 182], [158, 182],
		[155, 180], [148, 185], [162, 185], [155, 188],
	) unless @ok;
	my $s = $ok[int(rand(scalar @ok))];
	return ($s->[0], $s->[1]);
}

sub Phase1_WalkFountain {
	return 0 unless Phase1_IsInPronteraTown();
	if ($field && $field->baseName eq 'prt_in') {
		Commands::run("move prontera 156 190");
		return 1;
	}
	my ($x, $y) = Phase1_RandomFountainTile();
	Log::message("[PHASE1] fountain walk → $x,$y\n", "success");
	Commands::run("stand") if $char && $char->{sitting};
	Commands::run("move $x $y");
	return 1;
}

# Body dirs: 0=N 1=NW 2=W 3=SW 4=S 5=SE 6=E 7=NE
# Walking one tile is the only reliable way other clients see facing.
sub Phase1_StepFace {
	return 0 unless $char && $field && $field->baseName eq 'prontera';
	Commands::run("stand") if $char->{sitting};
	AI::clear("move", "route", "mapRoute", "sitting", "standing", "look");

	my @dx = (0, -1, -1, -1, 0, 1, 1, 1);
	my @dy = (1, 1, 0, -1, -1, -1, 0, 1);
	my @dir = qw(N NW W SW S SE E NE);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;

	my @cands;
	for my $d (0 .. 7) {
		my ($tx, $ty) = ($x + $dx[$d], $y + $dy[$d]);
		next unless $ty >= 170 && abs($tx - 156) <= 22;
		push @cands, [$d, $tx, $ty];
	}
	unless (@cands) {
		for my $d (0 .. 7) {
			push @cands, [$d, $x + $dx[$d], $y + $dy[$d]];
		}
	}
	for (my $i = @cands - 1; $i > 0; $i--) {
		my $j = int(rand($i + 1));
		@cands[$i, $j] = @cands[$j, $i];
	}
	my ($body, $tx, $ty) = @{$cands[0]};
	Commands::run("conf sitAuto_look $body");
	Commands::run("conf sitAuto_look_delay 0");
	Commands::run("move $tx $ty");
	Log::message("[PHASE1] AFK step face=$body ($dir[$body]) → $tx,$ty\n", "success");
	return $body;
}

sub Phase1_AfkSit {
	return 1 if $char && $char->{sitting};
	# Prevent processSitAuto from standing us immediately at full HP
	$ai_v{sitAuto_forcedBySitCommand} = 1;
	$ai_v{sitAuto_forceStop} = 0;
	Commands::run("sit");
	return 1;
}

sub Phase1_TalkJobMaster {
	my $job = lc($config{grindTargetJob} || 'random');
	my @pool = qw(Swordman Magician Archer Acolyte Merchant Thief);
	my %menu = (
		swordman => 'r0',
		swordsman => 'r0',
		magician => 'r1',
		mage => 'r1',
		archer => 'r2',
		acolyte => 'r3',
		merchant => 'r4',
		thief => 'r5',
	);
	if ($job eq '' || $job eq 'random' || $job eq 'none' || $job eq 'auto') {
		my $pick = $pool[int(rand(scalar @pool))];
		Commands::run("conf grindTargetJob $pick");
		$job = lc($pick);
		Log::message("[PHASE1] Random class selected: $pick\n", "success");
	}
	my $r = $menu{$job} // 'r0';
	Log::message("[PHASE1] Job Master talk for '$job' ($r) @ 155,180\n", "success");
	Commands::run("stand");
	Commands::run("talknpc 155 180 c c c $r c r0");
	return 1;
}

sub Phase1_AfterJobChange {
	my $base = $char ? ($char->{lv} // 0) : 0;
	if ($base >= 15) {
		Log::message("[PHASE1] Base $base + jobbed — park at fountain\n", "success");
		Commands::run("conf phase1Done 1");
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		Commands::run("conf sellAuto 0");
		return 1;
	}
	Log::message("[PHASE1] Base $base < 15 — resume prt_fild08 as 1st job\n", "success");
	Commands::run("conf -f grindHuntMap prt_fild08");
	Phase1_ResumeHunt();
	return 1;
}

sub Phase1_ResumeHunt {
	if ($config{phase1Afk} || $config{phase1Done}) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	if ($char && $char->{jobID} == 0 && $char->{lv_job} >= 10) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Commands::run("conf lockMap prt_fild08");
	Commands::run("conf lockMap_x none");
	Commands::run("conf -f grindHuntMap prt_fild08");
	Commands::run("conf attackAuto 2");
	Commands::run("conf attackAuto_inLockOnly 1");
	Commands::run("conf attackRouteMaxPathDistance 50");
	Commands::run("conf attackMinPlayerDistance 0");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf sellAuto 0");
	Log::message("[PHASE1] Resume hunt prt_fild08\n", "success");
	return 1;
}
