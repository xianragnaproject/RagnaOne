###############################################################################
# 7) Stay online — RUN ONCE (macro-owned reconnect; no direct timeouts.txt edit)
###############################################################################
automacro Phase1_StayOnline {
	exclusive 1
	timeout 20
	priority 0
	ConfigKeyNot phase1StayOnline 1
	call {
		[
			log PHASE1: stay-online reconnect (run once)
			do conf dcPause 0
			do conf dcOnDeath 0
			do conf dcOnDisconnect 0
			do conf dcOnServerShutDown 0
			do conf dcOnServerClose 0
			do conf dcOnMaxReconnections 0
			do conf autoRestart 0
			do timeout reconnect 10
			do timeout reconnect_backoff 10,20,30,60,90,120
			do timeout reconnect_random 5
			do timeout play 20
			do conf -f phase1StayOnline 1
		]
	}
}

macro phase1_sell_loot {

		log PHASE1: OW>=50% - sell at Prontera tool dealer
		do conf phase1Selling 1
		do conf attackAuto 0
		do conf route_randomWalk 0
		do conf lockMap none
		do conf sellAuto 0
		do conf sellAuto_npc prt_in 126 76
		do conf sellAuto_standpoint prt_in 130 72
		do stand
		do eval AI::clear("move", "route", "mapRoute", "sellAuto")
	]
	call phase1_to_prontera
	[
		do move prt_in 130 72
		pause 45
		do conf sellAuto 1
		do eval AI::queue("sellAuto")
		pause 35
		do conf sellAuto 0
		do move 135 71
		pause 6
		do move prontera 160 70
		pause 10
		do conf phase1Selling 0
		do eval Phase1_ResumeHunt()
		log PHASE1: sell done - resume hunt
	]
}

###############################################################################

macro phase1_to_prontera {
	[
		log PHASE1: go Prontera
		do conf lockMap none
		do conf route_randomWalk 0
		do conf attackAuto 0
		do stand
		do eval AI::clear("move", "route", "mapRoute", "attack")
		do eval Phase1_GoProntera()
	]
	if (&config(phase1InProntera) == 1) goto phase1_town_ready
	pause 70
	[
		do stand
		do eval Phase1_GoProntera()
	]
	if (&config(phase1InProntera) == 1) goto phase1_town_ready
	pause 50
	[
		do stand
		do eval Phase1_GoProntera()
	]
	pause 50
	:phase1_town_ready
	[
		log PHASE1: Prontera ready
	]
}

macro phase1_to_payon {
	[
		log PHASE1: go Payon via Kafra (no field walk)
		do conf lockMap none
		do conf route_randomWalk 0
		do conf attackAuto 0
		do stand
		do eval Phase1_GoPayonTown()
	]
	if (&config(phase1InPayon) == 1) goto phase1_payon_ready
	pause 40
	[
		do eval Phase1_GoPayonTown()
	]
	if (&config(phase1InPayon) == 1) goto phase1_payon_ready
	pause 45
	[
		do eval Phase1_GoPayonTown()
	]
	if (&config(phase1InPayon) == 1) goto phase1_payon_ready
	pause 45
	[
		do eval Phase1_GoPayonTown()
	]
	pause 30
	if (&config(phase1InPayon) == 1) goto phase1_payon_ready
	[
		log PHASE1: still not in Payon - retry next tick
		do conf phase1Traveling 0
	]
	stop
	:phase1_payon_ready
	[
		log PHASE1: Payon ready
		do conf phase1Traveling 0
	]
}

sub Phase1_IsInPronteraTown {
	return 0 unless $field;
	my $m = $field->baseName // '';
	return ($m eq 'prontera' || $m eq 'prt_in') ? 1 : 0;
}

sub Phase1_GoProntera {
	if (Phase1_IsInPronteraTown()) {
		Commands::run("conf -f phase1InProntera 1");
		return 1;
	}
	Commands::run("conf -f phase1InProntera 0");
	return 1 unless $field && $char;
	Commands::run("stand") if $char->{sitting};
	Log::message("[PHASE1] map-route → Prontera fountain\n", "success");
	Commands::run("move prontera 156 190");
	return 1;
}

sub Phase1_TalkJobMaster {
	my @pool = qw(Swordman Magician Archer Acolyte Merchant Thief);
	my %menu = (
		swordman => 'r0',
		swordsman => 'r0',
		magician => 'r1',
		mage => 'r1',
		archer => 'r2',
		acolyte => 'r3',
		merchant => 'r4',
		thief => 'r5',
	);
	my $pick = $pool[int(rand(scalar @pool))];
	Commands::run("conf -f grindTargetJob $pick");
	my $r = $menu{lc($pick)} // 'r0';
	Log::message("[PHASE1] random class $pick ($r) @ Job Master 155,180\n", "success");
	Commands::run("stand");
	Commands::run("talknpc 155 180 c c c $r c r0");
	return 1;
}

sub Phase1_ClearStuckJobbing {
	return 0 unless ($config{phase1Jobbing} // 0);
	my $jobbed = $char && ($char->{jobID} // 0) != 0;
	my $ready = $char
		&& ($char->{jobID} // 0) == 0
		&& ($char->{lv_job} // 0) >= 10
		&& ($char->{lv} // 0) >= 15;
	if ($jobbed || !$ready) {
		Log::message("[PHASE1] clear stuck phase1Jobbing\n", "system");
		Commands::run("conf phase1Jobbing 0");
		Phase1_ResumeHunt() unless $config{phase1Done};
		return 1;
	}
	return 0;
}

sub Phase1_IsInPayonTown {
	return 0 unless $field;
	return (($field->baseName // '') eq 'payon') ? 1 : 0;
}

# Prontera Kafra Teleport Service → Payon (c r2 c r2)
# Kafras: south 151,29 | mid 146,89 | east 282,200 | west 35,208
sub Phase1_TalkKafraWarpPayon {
	my @kafras = (
		[151, 29],
		[146, 89],
		[282, 200],
		[35, 208],
	);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;
	my ($best, $bd);
	foreach my $k (@kafras) {
		my $d = abs($x - $k->[0]) + abs($y - $k->[1]);
		if (!defined $bd || $d < $bd) {
			$bd = $d;
			$best = $k;
		}
	}
	return 0 unless $best;
	if ($bd > 16) {
		Log::message("[PHASE1] walk to Kafra $best->[0],$best->[1]\n", "success");
		Commands::run("move $best->[0] $best->[1]");
		return 0;
	}
	Log::message("[PHASE1] Teleport Service → Payon via $best->[0],$best->[1]\n", "success");
	Commands::run("talknpc $best->[0] $best->[1] c r2 c r2");
	return 1;
}

sub Phase1_GoPayonTown {
	Commands::run("stand") if $char && $char->{sitting};
	my $m = $field ? ($field->baseName // '') : '';
	if ($m eq 'payon') {
		Commands::run("conf -f phase1InPayon 1");
		return 1;
	}
	Commands::run("conf -f phase1InPayon 0");
	if (AI::inQueue("route") || AI::inQueue("mapRoute") || AI::inQueue("move")
		|| AI::is("route", "mapRoute", "move") || AI::is("NPC")) {
		return 0;
	}
	if ($m eq 'prontera') {
		return Phase1_TalkKafraWarpPayon();
	}
	if ($m eq 'prt_fild08' || $m eq 'prt_in') {
		Phase1_GoProntera();
		return 0;
	}
	Commands::run("move prontera 156 190");
	return 0;
}

sub Phase1_ResumeHunt {
	return 1 if $config{phase1Done} || $config{phase1Jobbing};
	if ($char && $char->{jobID} == 0 && ($char->{lv_job} // 0) >= 10 && ($char->{lv} // 0) >= 15) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Commands::run("conf lockMap prt_fild08");
	Commands::run("conf lockMap_x none");
	Commands::run("conf attackAuto 2");
	Commands::run("conf attackAuto_inLockOnly 1");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf sellAuto 0");
	Log::message("[PHASE1] resume hunt prt_fild08\n", "success");
	return 1;
}
