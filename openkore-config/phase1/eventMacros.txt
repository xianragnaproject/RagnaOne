macro phase1_to_prontera {
	[
		log PHASE1: go Prontera for town chores
		do conf lockMap none
		do conf route_randomWalk 0
		do conf attackAuto 0
		do conf sitAuto_idle 0
		do stand
		do eval AI::clear("move", "route", "mapRoute", "attack")
		do eval Phase1_GoPronteraPortal()
	]
	# Already in town? skip multi-minute portal waits (avoids GM idle kicks)
	if (&config(phase1InProntera) == 1) goto phase1_town_ready
	pause 90
	[
		do stand
		do eval Phase1_GoPronteraPortal()
	]
	if (&config(phase1InProntera) == 1) goto phase1_town_ready
	pause 60
	[
		do stand
		do eval Phase1_ForceProntera()
	]
	pause 60
	:phase1_town_ready
	[
		log PHASE1: Prontera ready
	]
}

sub Phase1_ForceProntera {
	return 1 if Phase1_IsInPronteraTown();
	Commands::run("stand") if $char && $char->{sitting};
	Log::message("[PHASE1] force map-route → prontera 156,190\n", "success");
	Commands::run("move prontera 156 190");
	return 1;
}


###############################################################################
# Perl helpers
###############################################################################
sub Phase1_IsInPronteraTown {
	return 0 unless $field;
	my $m = $field->baseName;
	return ($m eq 'prontera' || $m eq 'prt_in') ? 1 : 0;
}

sub Phase1_ApproachFieldPortal {
	return 1 if Phase1_IsInPronteraTown();
	return 1 unless $field && $field->baseName eq 'prt_fild08';
	my $x = $char->{pos}{x} // 0;
	my $y = $char->{pos}{y} // 0;
	return 1 if abs($x - 170) <= 2 && abs($y - 375) <= 3;
	Commands::run("move 170 375");
	return 1;
}

# Combined portal helper: approach + enter. Safe to call repeatedly.
sub Phase1_GoPronteraPortal {
	if (Phase1_IsInPronteraTown()) {
		Commands::run("conf -f phase1InProntera 1");
		return 1;
	}
	Commands::run("conf -f phase1InProntera 0");
	return 1 unless $field && $char;
	Commands::run("stand") if $char->{sitting};
	# Use OpenKore map-route (knows portals) — more reliable than manual lip bump
	Log::message("[PHASE1] map-route → Prontera fountain area\n", "success");
	Commands::run("move prontera 156 190");
	return 1;
}

sub Phase1_EnterFieldPortal {
	return Phase1_GoPronteraPortal();
}

sub Phase1_RandomFountainTile {
	my @spots = (
		[155, 185], [150, 185], [160, 185], [148, 185], [162, 185],
		[155, 180], [152, 182], [158, 182], [150, 182], [160, 182],
		[155, 188], [152, 188], [158, 188], [148, 188], [162, 188],
		[155, 190], [150, 190], [160, 190], [152, 190], [158, 190],
		[145, 192], [146, 188], [166, 192], [167, 188],
		[145, 185], [167, 185],
	);
	my @ok;
	if ($field && $field->baseName eq 'prontera') {
		for my $s (@spots) {
			next unless $field->isWalkable($s->[0], $s->[1]);
			push @ok, $s;
		}
	}
	@ok = (
		[155, 185], [150, 185], [160, 185], [152, 182], [158, 182],
		[155, 180], [148, 185], [162, 185], [155, 188],
	) unless @ok;
	my $s = $ok[int(rand(scalar @ok))];
	return ($s->[0], $s->[1]);
}

sub Phase1_WalkFountain {
	unless (Phase1_IsInPronteraTown()) {
		Log::message("[PHASE1] fountain walk skipped - not in Prontera yet\n", "system");
		return 0;
	}
	if ($field && $field->baseName eq 'prt_in') {
		Commands::run("move prontera 156 190");
		return 1;
	}
	my ($x, $y) = Phase1_RandomFountainTile();
	Log::message("[PHASE1] fountain walk → $x,$y\n", "success");
	Commands::run("stand") if $char && $char->{sitting};
	Commands::run("move $x $y");
	return 1;
}

# Body dirs: 0=N 1=NW 2=W 3=SW 4=S 5=SE 6=E 7=NE
# Walking one tile is the only reliable way other clients see facing.
sub Phase1_StepFace {
	unless ($char && $field && $field->baseName eq 'prontera') {
		Log::message("[PHASE1] AFK face skipped - not in Prontera\n", "system");
		return 0;
	}
	Commands::run("stand") if $char->{sitting};
	AI::clear("move", "route", "mapRoute", "sitting", "standing", "look");

	my @dx = (0, -1, -1, -1, 0, 1, 1, 1);
	my @dy = (1, 1, 0, -1, -1, -1, 0, 1);
	my @dir = qw(N NW W SW S SE E NE);
	my $x = $char->{pos_to}{x} // $char->{pos}{x} // 0;
	my $y = $char->{pos_to}{y} // $char->{pos}{y} // 0;

	my @cands;
	for my $d (0 .. 7) {
		my ($tx, $ty) = ($x + $dx[$d], $y + $dy[$d]);
		next unless $ty >= 170 && abs($tx - 156) <= 22;
		push @cands, [$d, $tx, $ty];
	}
	unless (@cands) {
		for my $d (0 .. 7) {
			push @cands, [$d, $x + $dx[$d], $y + $dy[$d]];
		}
	}
	for (my $i = @cands - 1; $i > 0; $i--) {
		my $j = int(rand($i + 1));
		@cands[$i, $j] = @cands[$j, $i];
	}
	my ($body, $tx, $ty) = @{$cands[0]};
	Commands::run("conf sitAuto_look $body");
	Commands::run("conf sitAuto_look_delay 0");
	Commands::run("move $tx $ty");
	Log::message("[PHASE1] AFK step face=$body ($dir[$body]) → $tx,$ty\n", "success");
	return $body;
}

sub Phase1_AfkSit {
	return 1 if $char && $char->{sitting};
	unless (Phase1_IsInPronteraTown()) {
		Log::message("[PHASE1] AFK sit skipped - not in Prontera\n", "system");
		return 0;
	}
	# This server requires Basic Skill >= 3 to sit
	my $basic = 0;
	if ($char && $char->{skills}{NV_BASIC}) {
		$basic = $char->{skills}{NV_BASIC}{lv} // 0;
	}
	if ($basic < 3) {
		Log::message("[PHASE1] AFK stand-idle - Basic Skill $basic < 3 cannot sit\n", "system");
		Commands::run("stand") if $char && $char->{sitting};
		return 0;
	}
	# Prevent processSitAuto from standing us immediately at full HP
	$ai_v{sitAuto_forcedBySitCommand} = 1;
	$ai_v{sitAuto_forceStop} = 0;
	Commands::run("sit");
	return 1;
}

# Small random walk near Prontera fountain — prevents GM idle kicks while "parked"
sub Phase1_AntiIdleStep {
	return 0 unless $char && Phase1_IsInPronteraTown();
	Commands::run("stand") if $char->{sitting};
	AI::clear("sitting", "standing");
	my $cx = $char->{pos_to}{x} // $char->{pos}{x} // 156;
	my $cy = $char->{pos_to}{y} // $char->{pos}{y} // 190;
	my $tx = $cx + int(rand(7)) - 3;
	my $ty = $cy + int(rand(7)) - 3;
	$tx = 145 if $tx < 145;
	$tx = 168 if $tx > 168;
	$ty = 175 if $ty < 175;
	$ty = 205 if $ty > 205;
	Commands::run("move $tx $ty");
	Commands::run("look " . int(rand(8)));
	Log::message("[PHASE1] anti-idle step → $tx,$ty\n", "success");
	return 1;
}

# Spend skill points until Basic Skill >= 3 so AFK sit works
sub Phase1_EnsureBasicSkillForSit {
	return 1 unless $char;
	my $basic = 0;
	if ($char->{skills}{NV_BASIC}) {
		$basic = $char->{skills}{NV_BASIC}{lv} // 0;
	}
	return 1 if $basic >= 3;
	Commands::run("stand") if $char->{sitting};
	my $points = $char->{points_skill} // 0;
	Log::message("[PHASE1] Raising Basic Skill for sit ($basic -> 3), free points=$points\n", "success");
	my $guard = 0;
	while ($basic < 3 && $guard++ < 5) {
		last if ($char->{points_skill} // 0) <= 0;
		Commands::run("skills add 1");
		select(undef, undef, undef, 0.4);
		$basic = ($char->{skills}{NV_BASIC}{lv} // 0) if $char->{skills}{NV_BASIC};
	}
	return 1;
}

sub Phase1_TalkJobMaster {
	my $job = lc($config{grindTargetJob} || 'random');
	my @pool = qw(Swordman Magician Archer Acolyte Merchant Thief);
	my %menu = (
		swordman => 'r0',
		swordsman => 'r0',
		magician => 'r1',
		mage => 'r1',
		archer => 'r2',
		acolyte => 'r3',
		merchant => 'r4',
		thief => 'r5',
	);
	if ($job eq '' || $job eq 'random' || $job eq 'none' || $job eq 'auto') {
		my $pick = $pool[int(rand(scalar @pool))];
		Commands::run("conf grindTargetJob $pick");
		$job = lc($pick);
		Log::message("[PHASE1] Random class selected: $pick\n", "success");
	}
	my $r = $menu{$job} // 'r0';
	Log::message("[PHASE1] Job Master talk for '$job' ($r) @ 155,180\n", "success");
	Commands::run("stand");
	Commands::run("talknpc 155 180 c c c $r c r0");
	return 1;
}

sub Phase1_ClearStuckJobbing {
	return 0 unless ($config{phase1Jobbing} // 0);
	my $jobbed = $char && ($char->{jobID} // 0) != 0;
	my $ready = $char
		&& ($char->{jobID} // 0) == 0
		&& ($char->{lv_job} // 0) >= 10
		&& ($char->{lv} // 0) >= 15;
	if ($jobbed || !$ready) {
		Log::message("[PHASE1] Clear stuck phase1Jobbing (jobbed=$jobbed ready=$ready)\n", "system");
		Commands::run("conf phase1Jobbing 0");
		Commands::run("conf phase1Afk 0") if $config{phase1Afk};
		Phase1_ResumeHunt() unless $config{phase1Done};
		return 1;
	}
	return 0;
}

sub Phase1_ApplyJobBuild {
	# Map jobID → class name when grindTargetJob not set
	my %id_to_job = (
		1 => 'swordman', 2 => 'magician', 3 => 'archer',
		4 => 'acolyte', 5 => 'merchant', 6 => 'thief',
	);
	my $job = lc($config{grindTargetJob} || '');
	if ($job eq '' || $job eq 'random' || $job eq 'none' || $job eq 'auto') {
		$job = $id_to_job{ $char->{jobID} // -1 } // 'swordman';
		Commands::run("conf grindTargetJob $job");
	}
	$job = 'swordman' if $job eq 'swordsman';
	$job = 'magician' if $job eq 'mage';

	my %skills = (
		swordman => 'Basic Skill 9, Bash 10, Provoke 5, Endure 1',
		magician => 'Basic Skill 9, Cold Bolt 5, Fire Bolt 5, Lightning Bolt 5, Increase SP Recovery 5',
		archer   => 'Basic Skill 9, Owl\'s Eye 10, Vulture\'s Eye 5, Double Strafe 5',
		acolyte  => 'Basic Skill 9, Divine Protection 5, Heal 10, Increase AGI 5, Blessing 5',
		merchant => 'Basic Skill 9, Enlarge Weight Limit 5, Discount 5, Overcharge 5, Pushcart 4',
		thief    => 'Basic Skill 9, Double Attack 10, Improve Dodge 10, Envenom 5',
	);
	my %stats = (
		swordman => '15 str, 15 dex, 15 agi, 25 dex, 15 vit, 30 str, 30 agi, 20 vit, 40 str, 40 agi, 35 dex',
		magician => '15 dex, 15 int, 20 dex, 40 int, 30 dex, 50 int, 20 vit, 60 int, 40 dex',
		archer   => '15 dex, 15 agi, 30 dex, 30 agi, 40 dex, 40 agi, 50 dex, 50 agi, 60 dex',
		acolyte  => '15 str, 15 dex, 25 str, 25 dex, 15 vit, 30 dex, 40 str, 20 vit, 20 int, 50 str',
		merchant => '15 dex, 15 str, 20 agi, 25 str, 25 dex, 30 vit, 40 str, 30 dex, 40 agi',
		thief    => '10 str, 10 agi, 20 dex, 20 str, 40 agi, 30 dex, 30 str, 50 agi, 40 dex, 60 agi',
	);

	my $sk = $skills{$job} // $skills{swordman};
	my $st = $stats{$job} // $stats{swordman};
	Commands::run("conf -f skillsAddAuto_list $sk");
	Commands::run("conf -f skillsAddAuto 1");
	Commands::run("conf -f statsAddAuto_list $st");
	Commands::run("conf -f statsAddAuto 1");
	Log::message("[PHASE1] Auto skill/stat build for $job\n", "success");
	return 1;
}

sub Phase1_AfterJobChange {
	my $base = $char ? ($char->{lv} // 0) : 0;
	# Always keep hunting after job — no fountain park (idle = GM kick on this server)
	Log::message("[PHASE1] Base $base + jobbed - keep hunting prt_fild08 (no idle park)\n", "success");
	Commands::run("conf phase1Done 1");
	Commands::run("conf phase1Afk 0");
	Commands::run("conf sitAuto_idle 0");
	Commands::run("conf -f grindHuntMap prt_fild08");
	Commands::run("conf lockMap prt_fild08");
	Commands::run("conf lockMap_x none");
	Commands::run("conf lockMap_y none");
	Commands::run("conf attackAuto 2");
	Commands::run("conf attackAuto_inLockOnly 1");
	Commands::run("conf attackRouteMaxPathDistance 50");
	Commands::run("conf attackMinPlayerDistance 0");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf route_randomWalk_inLockOnly 1");
	Commands::run("conf itemsTakeAuto 2");
	Commands::run("conf itemsGatherAuto 2");
	Commands::run("conf sellAuto 0");
	Commands::run("stand");
	Commands::run("ai on");
	return 1;
}

sub Phase1_ResumeHunt {
	# Never leave bots idle — phase1Done still hunts
	if ($config{phase1Afk}) {
		Commands::run("conf phase1Afk 0");
	}
	# Ready for job change — stop hunting so job macros can run
	if ($char && $char->{jobID} == 0 && ($char->{lv_job} // 0) >= 10 && ($char->{lv} // 0) >= 15
		&& !($config{phase1JobDone} // 0)) {
		Commands::run("conf attackAuto 0");
		Commands::run("conf route_randomWalk 0");
		Commands::run("conf lockMap none");
		return 1;
	}
	Commands::run("conf lockMap prt_fild08");
	Commands::run("conf lockMap_x none");
	Commands::run("conf -f grindHuntMap prt_fild08");
	Commands::run("conf attackAuto 2");
	Commands::run("conf attackAuto_inLockOnly 1");
	Commands::run("conf attackRouteMaxPathDistance 50");
	Commands::run("conf attackMinPlayerDistance 0");
	Commands::run("conf route_randomWalk 1");
	Commands::run("conf route_randomWalk_inLockOnly 1");
	Commands::run("conf sellAuto 0");
	Commands::run("conf sitAuto_idle 0");
	Log::message("[PHASE1] Resume hunt prt_fild08\n", "success");
	return 1;
}
